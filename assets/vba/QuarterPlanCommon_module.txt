Public Sub HandleQuarterPlanShapeAction(ByVal shapeName As String)
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim shp As Shape
    Dim actionCell As Range
    Dim normalizedShapeName As String
    Dim actionHandled As Boolean

    If mQuarterPlanUiBusy Or mQuarterPlanActionRunning Then Exit Sub

    Set ws = GetQuarterPlanWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Quarter plan worksheet not found by tblPlanActive."
    normalizedShapeName = NormalizeShapeCallerName(shapeName)
    Set shp = ws.Shapes(normalizedShapeName)
    Set actionCell = shp.TopLeftCell

    mQuarterPlanActionRunning = True
    Application.EnableEvents = False

    If actionCell.MergeCells Then Set actionCell = actionCell.MergeArea.Cells(1, 1)

    If actionCell.Address(False, False) = QP_BACKLOG_ACTION_CELL And Trim$(CStr(actionCell.Value)) = QuarterPlanTextAddAllToPlan() Then
        Application.ScreenUpdating = False
        RunQuarterPlanAddAllBacklogToPlan
    ElseIf IsQuarterPlanActionCell(ws, actionCell) Then
        Application.ScreenUpdating = False
        actionHandled = HandleQuarterPlanAction(ws, actionCell)
    End If

SafeExit:
    Application.ScreenUpdating = True
    mQuarterPlanActionRunning = False
    Application.EnableEvents = True
    Exit Sub

Fail:
    If Left$(normalizedShapeName, Len(QUARTER_PLAN_CELL_ACTION_LEGACY_PREFIX)) = QUARTER_PLAN_CELL_ACTION_LEGACY_PREFIX Then
        On Error Resume Next
        RebuildQuarterPlanActionUi
        On Error GoTo 0
        MsgBox "Кнопки действий были созданы старой версией с длинными именами и Excel вернул обрезанное имя." & vbCrLf & _
            "Интерфейс кнопок пересоздан. Нажмите действие еще раз." & vbCrLf & _
            "Caller: " & shapeName & vbCrLf & _
            "Shape: " & normalizedShapeName, vbInformation
        Resume SafeExit
    End If

    MsgBox "Ошибка HandleQuarterPlanShapeAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Caller: " & shapeName & vbCrLf & _
        "Shape: " & normalizedShapeName, vbExclamation
    Resume SafeExit
End Sub

Public Sub HandleTaskEstimateShapeAction(ByVal shapeName As String)
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim shp As Shape
    Dim actionCell As Range
    Dim normalizedShapeName As String

    If mQuarterPlanUiBusy Or mQuarterPlanActionRunning Then Exit Sub

    Set ws = GetTaskEstimatesWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    normalizedShapeName = NormalizeShapeCallerName(shapeName)
    Set shp = ws.Shapes(normalizedShapeName)
    Set actionCell = shp.TopLeftCell

    mQuarterPlanActionRunning = True
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    If Not IsTaskEstimateActionCell(ws, actionCell) Then
        Err.Raise 5, , "Clicked shape is not attached to a valid task estimate action cell: " & actionCell.Address(False, False)
    End If
    If HandleTaskEstimateAction(ws, actionCell) Then
        RefreshTaskEstimateActionGrid
    End If

SafeExit:
    Application.ScreenUpdating = True
    mQuarterPlanActionRunning = False
    Application.EnableEvents = True
    Exit Sub

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "HandleTaskEstimateShapeAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Caller: " & shapeName & vbCrLf & _
        "Shape: " & normalizedShapeName, vbExclamation
    Resume SafeExit
End Sub

Public Function NormalizeShapeCallerName(ByVal callerName As String) As String
    Dim result As String
    Dim bangPos As Long

    result = Trim$(callerName)
    result = Replace(result, "'", "")

    bangPos = InStrRev(result, "!")
    If bangPos > 0 Then
        result = Mid$(result, bangPos + 1)
    End If

    NormalizeShapeCallerName = result
End Function

Public Function FindWorksheetByTableName(ByVal tableName As String) As Worksheet
    Dim ws As Worksheet
    Dim tbl As ListObject

    For Each ws In ThisWorkbook.Worksheets
        Set tbl = Nothing
        On Error Resume Next
        Set tbl = ws.ListObjects(tableName)
        On Error GoTo 0
        If Not tbl Is Nothing Then
            Set FindWorksheetByTableName = ws
            Exit Function
        End If
    Next ws
End Function

Public Function WorksheetHasTable(ByVal ws As Worksheet, ByVal tableName As String) As Boolean
    Dim tbl As ListObject

    If ws Is Nothing Then Exit Function
    On Error Resume Next
    Set tbl = ws.ListObjects(tableName)
    On Error GoTo 0
    WorksheetHasTable = Not tbl Is Nothing
End Function

Public Sub AssertTaskEstimateUnicodeTextForTest()
    Dim headers As Variant
    Dim fileName As String

    headers = TaskEstimateVisibleHeaders()
    If CStr(headers(0)) <> TaskEstimateHeaderPriority() Or CStr(headers(10)) <> TaskEstimateTextExportXlsx() Then
        Err.Raise 5, "AssertTaskEstimateUnicodeTextForTest", "Task estimate header helpers are inconsistent."
    End If
    If TaskEstimateTextStatistics() <> TextFromCodes(1057, 1090, 1072, 1090, 1080, 1089, 1090, 1080, 1082, 1072) Then
        Err.Raise 5, "AssertTaskEstimateUnicodeTextForTest", "Task estimate statistics text is inconsistent."
    End If
    If TaskEstimateCsvHeaderKey() = "" Or TaskEstimateCsvHeaderTheme() = "" Or TaskEstimateCsvHeaderDirection() = "" Or TaskEstimateTextClearPrompt() = "" Or TaskEstimateTextImportCsvTitle() = "" Then
        Err.Raise 5, "AssertTaskEstimateUnicodeTextForTest", "Task estimate dialog helpers are empty."
    End If

    fileName = DefaultTaskEstimateExportFileName()
    If InStr(1, fileName, TaskEstimateFilePartEstimateTasks(), vbBinaryCompare) = 0 Then
        Err.Raise 5, "AssertTaskEstimateUnicodeTextForTest", "Task estimate default filename has no Unicode-safe estimate suffix: " & fileName
    End If
    If LCase$(Right$(fileName, 5)) <> ".xlsx" Then
        Err.Raise 5, "AssertTaskEstimateUnicodeTextForTest", "Task estimate default filename does not end with .xlsx: " & fileName
    End If
End Sub

Public Function DefaultQuarterPlanBacklogExportFileName() As String
    Dim wsSettings As Worksheet
    Dim teamName As String
    Dim yearValue As String
    Dim quarterValue As String

    On Error Resume Next
    Set wsSettings = FindWorksheetByTableName("tblTeamComposition")
    If Not wsSettings Is Nothing Then
        teamName = Trim$(CStr(wsSettings.Range("B4").Value))
        yearValue = Trim$(CStr(wsSettings.Range("B6").Value))
        quarterValue = Trim$(CStr(wsSettings.Range("B7").Value))
    End If
    On Error GoTo 0

    If teamName = "" Then teamName = "Команда"
    If yearValue = "" Then yearValue = Format$(Date, "yyyy")
    If quarterValue = "" Then quarterValue = "Q"

    DefaultQuarterPlanBacklogExportFileName = SanitizeFileName(teamName & "_" & yearValue & "_Q" & quarterValue & "_" & QuarterPlanFilePartQuarterlyPlan() & "_" & Format$(Now, "yyyy-mm-dd_hh-nn") & ".xlsx")
End Function

Public Function EnsureXlsxExtension(ByVal filePath As String) As String
    If filePath = "" Then Exit Function
    If LCase$(Right$(filePath, 5)) <> ".xlsx" Then filePath = filePath & ".xlsx"
    EnsureXlsxExtension = filePath
End Function

Public Function SanitizeFileName(ByVal fileName As String) As String
    Dim invalidChars As Variant
    Dim item As Variant

    invalidChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For Each item In invalidChars
        fileName = Replace(fileName, CStr(item), "_")
    Next item
    SanitizeFileName = Trim$(fileName)
End Function
