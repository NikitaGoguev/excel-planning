Public Sub ApplyDateFormat(ByVal targetRange As Range)
    targetRange.NumberFormatLocal = DateFormatLocalText()
End Sub

Public Function DateFormatLocalText() As String
    DateFormatLocalText = ChrW$(&H414) & ChrW$(&H414) & "-" & _
                          ChrW$(&H41C) & ChrW$(&H41C) & "-" & _
                          ChrW$(&H413) & ChrW$(&H413) & ChrW$(&H413) & ChrW$(&H413)
End Function

    ws.Columns(PLAN_TECH_FIRST_COL_LETTER & ":" & PLAN_TECH_LAST_COL_LETTER).Hidden = False

End Sub

Public Sub RefreshTaskEstimateActionGrid()
    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim restoredTemplate As Boolean

    On Error GoTo Fail
    mQuarterPlanClosing = False
    mQuarterPlanUiBusy = True
    Set ws = GetTaskEstimatesWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    Set tbl = ws.ListObjects("tblTaskEstimates")

    restoredTemplate = RestoreTaskEstimateTemplate(ws, tbl)
    If restoredTemplate Then
        RemoveTaskEstimateActionButtons ws
        mTaskEstimateMaxActionRow = 0
        MarkTaskEstimateRefreshFrom 1
    End If

    RefreshTaskEstimateTable tbl
    AddTaskEstimateTopButtons ws
    SyncTaskEstimateActionButtons ws

SafeExit:
    mQuarterPlanUiBusy = False
    Exit Sub

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "RefreshTaskEstimateActionGrid: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
    Resume SafeExit
End Sub

Public Sub SetupTaskEstimateActionUi(ByVal ws As Worksheet)
    Dim tbl As ListObject
    Dim restoredTemplate As Boolean

    On Error GoTo Fail

    Set tbl = ws.ListObjects("tblTaskEstimates")
    If TaskEstimateTemplateNeedsRestore(ws, tbl) Then
        restoredTemplate = RestoreTaskEstimateTemplate(ws, tbl)
        RemoveTaskEstimateActionButtons ws
        mTaskEstimateMaxActionRow = 0
        MarkTaskEstimateRefreshFrom 1
    End If

    If restoredTemplate Or mTaskEstimateRefreshStartRow > 0 Then
        RefreshTaskEstimateTable tbl
    End If
    AddTaskEstimateTopButtons ws
    SyncTaskEstimateActionButtons ws
    Exit Sub

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "SetupTaskEstimateActionUi: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
End Sub

Public Function TaskEstimateTemplateNeedsRestore(ByVal ws As Worksheet, ByVal tbl As ListObject) As Boolean
    Dim targetRange As Range

    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet is not available."
    If tbl Is Nothing Then Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl Is Nothing Then Err.Raise 9, , "tblTaskEstimates not found."

    If tbl.Range.Row <> QP_TASK_HEADER_ROW Or tbl.Range.Column <> EST_COL_PRIORITY + 2 Then
        Err.Raise 5, , "tblTaskEstimates must start at C3 for template recovery."
    End If

    Set targetRange = ws.Range(QP_TASK_TABLE_RANGE)
    TaskEstimateTemplateNeedsRestore = (tbl.Range.Address(False, False) <> targetRange.Address(False, False))
End Function

Public Function RestoreTaskEstimateTemplate(ByVal ws As Worksheet, ByRef tbl As ListObject) As Boolean
    Dim targetRange As Range

    On Error GoTo Fail

    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet is not available."
    If tbl Is Nothing Then Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl Is Nothing Then Err.Raise 9, , "tblTaskEstimates not found."

    If tbl.Range.Row <> QP_TASK_HEADER_ROW Or tbl.Range.Column <> EST_COL_PRIORITY + 2 Then
        Err.Raise 5, , "tblTaskEstimates must start at C3 for template recovery."
    End If

    Set targetRange = ws.Range(QP_TASK_TABLE_RANGE)
    If tbl.Range.Address(False, False) <> targetRange.Address(False, False) Then
        tbl.Resize targetRange
        RestoreTaskEstimateTemplate = True
        Set tbl = ws.ListObjects("tblTaskEstimates")
    End If

    RestoreTaskEstimateTemplateHeader ws
    RestoreTaskEstimateTemplateStyles ws, tbl
    RestoreTaskEstimateTemplateValidation ws
    RestoreTaskEstimateTemplateStats ws
    Exit Function

Fail:
    Err.Raise Err.Number, "RestoreTaskEstimateTemplate", Err.Description
End Function

Public Sub RestoreTaskEstimateTemplateHeader(ByVal ws As Worksheet)
    Dim headers As Variant
    Dim index As Long

    headers = Array(">", "x", TaskEstimateHeaderPriority(), TaskEstimateHeaderDirection(), TaskEstimateHeaderDescription(), _
        TaskEstimateHeaderTicket(), TaskEstimateHeaderNote(), "AN", "BE", "FE", "QA", TaskEstimateHeaderComment(), TaskEstimateTextExportXlsx())
    For index = LBound(headers) To UBound(headers)
        ws.Cells(QP_TASK_HEADER_ROW, index + 1).Value = headers(index)
    Next index
    ws.Cells(QP_TASK_HEADER_ROW, EST_SHEET_COL_ACTION_COMMENT_OPTIONS).Value = "+"
End Sub

        .VerticalAlignment = xlCenter
        .WrapText = True
        .Borders.LineStyle = xlContinuous

    With actionRange

        .Font.Size = DESIGN_FONT_SIZE
        .Font.Bold = True
        .Font.Color = RGB(31, 78, 121)
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .WrapText = True
        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(128, 128, 128)
    End With

    With inputRange
        .Interior.Color = RGB(255, 242, 204)
        .Font.Name = DESIGN_FONT_NAME
        .Font.Size = DESIGN_FONT_SIZE
        .Font.Bold = False
        .Font.Color = RGB(31, 31, 31)

        .VerticalAlignment = xlCenter

    ws.Range("H" & QP_TASK_FIRST_ROW & ":K" & QP_TASK_LAST_ROW).NumberFormat = "0.00"

Public Sub RestoreTaskEstimateTemplateValidation(ByVal ws As Worksheet)
    With ws.Range("C" & QP_TASK_FIRST_ROW & ":C" & QP_TASK_LAST_ROW).Validation
        .Delete
        .Add Type:=xlValidateWholeNumber, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:="1", Formula2:="999"
        .IgnoreBlank = True
    End With

    With ws.Range("H" & QP_TASK_FIRST_ROW & ":K" & QP_TASK_LAST_ROW).Validation
        .Delete
        .Add Type:=xlValidateDecimal, AlertStyle:=xlValidAlertStop, Operator:=xlGreaterEqual, Formula1:="0"
        .IgnoreBlank = True
    End With
End Sub

Public Sub SyncTaskEstimateActionButtons(ByVal ws As Worksheet)
    On Error GoTo Fail

    Dim tbl As ListObject
    Dim rowIndex As Long
    Dim colIndex As Variant
    Dim actionCell As Range
    Dim shapeName As String
    Dim macroName As String
    Dim currentMaxRow As Long
    Dim syncMaxRow As Long
    Dim syncStartRow As Long

    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl.DataBodyRange Is Nothing Then Exit Sub

    currentMaxRow = LastTaskEstimateUsedRow(tbl)
    syncMaxRow = currentMaxRow + 1
    If mTaskEstimateMaxActionRow + 1 > syncMaxRow Then syncMaxRow = mTaskEstimateMaxActionRow + 1
    If syncMaxRow < 1 Then syncMaxRow = 1
    If syncMaxRow > tbl.DataBodyRange.Rows.Count Then syncMaxRow = tbl.DataBodyRange.Rows.Count
    syncStartRow = mTaskEstimateRefreshStartRow
    If syncStartRow < 1 Then syncStartRow = 1
    If syncStartRow > syncMaxRow Then syncStartRow = syncMaxRow

    For rowIndex = syncStartRow To syncMaxRow
        For Each colIndex In Array(EST_SHEET_COL_ACTION_DECOMPOSE, EST_SHEET_COL_ACTION_DELETE, EST_SHEET_COL_ACTION_EXPRESS_EXPORT, EST_SHEET_COL_ACTION_COMMENT_OPTIONS)
            Set actionCell = ws.Cells(tbl.DataBodyRange.Row + rowIndex - 1, CLng(colIndex))
            shapeName = TaskEstimateActionShapeName(rowIndex, CLng(colIndex))
            If Trim$(CStr(actionCell.Value)) = "" Then
                DeleteShapeIfExists ws, shapeName
            Else
                macroName = "RunTaskEstimateCellAction"
                If CLng(colIndex) = EST_SHEET_COL_ACTION_EXPRESS_EXPORT Then macroName = "RunTaskEstimateExpressExport"
                EnsureTaskEstimateButton ws, shapeName, actionCell, CStr(actionCell.Value), macroName
            End If
        Next colIndex
    Next rowIndex
    mTaskEstimateMaxActionRow = currentMaxRow
    mTaskEstimateRefreshStartRow = 0
    Exit Sub

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "SyncTaskEstimateActionButtons: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
End Sub

Public Function LastTaskEstimateUsedRow(ByVal tbl As ListObject) As Long
    Dim rowIndex As Long

    For rowIndex = tbl.DataBodyRange.Rows.Count To 1 Step -1
        If TaskEstimateRowHasTask(tbl, rowIndex) Then
            LastTaskEstimateUsedRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Public Sub MarkTaskEstimateRefreshFrom(ByVal rowIndex As Long)
    If rowIndex < 1 Then rowIndex = 1
    If mTaskEstimateRefreshStartRow = 0 Or rowIndex < mTaskEstimateRefreshStartRow Then
        mTaskEstimateRefreshStartRow = rowIndex
    End If
End Sub

Public Sub MarkTaskEstimateRefreshForTarget(ByVal ws As Worksheet, ByVal Target As Range)
    On Error GoTo SafeExit

    Dim tbl As ListObject
    Dim affected As Range
    Dim firstRow As Long

    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl Is Nothing Then Exit Sub
    If tbl.DataBodyRange Is Nothing Then Exit Sub

    Set affected = Intersect(Target, ws.Range("A" & QP_TASK_FIRST_ROW & ":N" & QP_TASK_LAST_ROW))
    If affected Is Nothing Then Exit Sub

    firstRow = affected.Row - tbl.DataBodyRange.Row + 1
    If firstRow < 1 Then firstRow = 1
    MarkTaskEstimateRefreshFrom firstRow

SafeExit:
End Sub

Public Function TaskEstimateIdInList(ByVal taskId As String, ByRef valueList() As String, ByVal valueCount As Long) As Boolean
    Dim index As Long
    Dim normalizedTaskId As String

    normalizedTaskId = Trim$(taskId)
    If normalizedTaskId = "" Then Exit Function

    For index = 1 To valueCount
        If valueList(index) = normalizedTaskId Then
            TaskEstimateIdInList = True
            Exit Function
        End If
    Next index
End Function

Public Sub RemoveStaleTaskEstimateActionButtons(ByVal ws As Worksheet, ByVal tbl As ListObject)
    Dim shapeIndex As Long
    Dim shp As Shape
    Dim actionCell As Range

    On Error Resume Next
    For shapeIndex = ws.Shapes.Count To 1 Step -1
        Set shp = ws.Shapes(shapeIndex)
        If Left$(shp.Name, Len(TASK_ESTIMATE_ACTION_SHAPE_PREFIX)) = TASK_ESTIMATE_ACTION_SHAPE_PREFIX Then
            Set actionCell = shp.TopLeftCell
            If actionCell.Row < tbl.DataBodyRange.Row _
                Or actionCell.Row > tbl.DataBodyRange.Row + tbl.DataBodyRange.Rows.Count - 1 _
                Or (actionCell.Column <> EST_SHEET_COL_ACTION_DECOMPOSE _
                    And actionCell.Column <> EST_SHEET_COL_ACTION_DELETE _
                    And actionCell.Column <> EST_SHEET_COL_ACTION_EXPRESS_EXPORT _
                    And actionCell.Column <> EST_SHEET_COL_ACTION_COMMENT_OPTIONS) _
                Or Trim$(CStr(actionCell.Value)) = "" Then
                shp.Delete
            End If
        End If
    Next shapeIndex
    On Error GoTo 0
End Sub

    Dim shapeName As String

    If tbl.DataBodyRange Is Nothing Then Exit Sub

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        For colIndex = 1 To PLAN_COL_ACTION_DOWN
            Set actionCell = tbl.DataBodyRange.Cells(rowIndex, colIndex)
            shapeName = QuarterPlanActionShapeName(tbl.Name, rowIndex, colIndex)
            If Trim$(CStr(actionCell.Value)) <> "" Then
                If Not QuarterPlanShapeExists(tbl.Parent, shapeName) Then
                    AddQuarterPlanButton tbl.Parent, shapeName, actionCell, CStr(actionCell.Value), _
                        "RunQuarterPlanCellAction"
                End If
            Else
                DeleteQuarterPlanShapeIfExists tbl.Parent, shapeName
            End If
        Next colIndex
    Next rowIndex

SafeExit:
End Sub

Public Function QuarterPlanShapeExists(ByVal ws As Worksheet, ByVal shapeName As String) As Boolean
    Dim shp As Shape

    On Error Resume Next
    Set shp = ws.Shapes(shapeName)
    QuarterPlanShapeExists = Not shp Is Nothing
    Set shp = Nothing
    On Error GoTo 0
End Function

Public Sub DeleteQuarterPlanShapeIfExists(ByVal ws As Worksheet, ByVal shapeName As String)
    On Error Resume Next
    ws.Shapes(shapeName).Delete
    On Error GoTo 0
End Sub

Public Function QuarterPlanActionShapeName(ByVal tableName As String, ByVal rowIndex As Long, ByVal colIndex As Long) As String
    Dim tableCode As String

    Select Case tableName
        Case "tblPlanActive"
            tableCode = "a"
        Case "tblPlanGrey"
            tableCode = "g"
        Case "tblPlanBacklog"
            tableCode = "b"
        Case Else
            tableCode = "x"
    End Select

    QuarterPlanActionShapeName = QUARTER_PLAN_CELL_ACTION_SHAPE_PREFIX & tableCode & "_" & Format$(rowIndex, "00") & "_" & CStr(colIndex)
End Function

Public Sub AddQuarterPlanButton(ByVal ws As Worksheet, ByVal shapeName As String, ByVal anchorCell As Range, ByVal caption As String, ByVal macroName As String)
    On Error GoTo SafeExit

    Dim anchorRange As Range
    Dim shp As Shape

    Set anchorRange = anchorCell
    If anchorRange.MergeCells Then Set anchorRange = anchorRange.MergeArea

    Set shp = ws.Shapes.AddShape(msoShapeRoundedRectangle, anchorRange.Left, anchorRange.Top, anchorRange.Width, anchorRange.Height)
    shp.Name = shapeName
    shp.OnAction = WorkbookMacroName(macroName)
    shp.Fill.ForeColor.RGB = RGB(31, 78, 121)
    shp.Line.ForeColor.RGB = RGB(31, 78, 121)
    shp.TextFrame.Characters.Text = caption
    shp.TextFrame.HorizontalAlignment = xlHAlignCenter
    shp.TextFrame.VerticalAlignment = xlVAlignCenter
    shp.TextFrame.Characters.Font.Name = DESIGN_FONT_NAME
    shp.TextFrame.Characters.Font.Size = DESIGN_FONT_SIZE
    shp.TextFrame.Characters.Font.Bold = True
    shp.TextFrame.Characters.Font.Color = RGB(255, 255, 255)
    shp.Placement = xlMoveAndSize
    shp.ZOrder msoBringToFront

SafeExit:
End Sub

Public Sub ButtonAddAllQuarterPlanBacklogToPlan()
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim sourceTbl As ListObject
    Dim targetTbl As ListObject
    Dim movedCount As Long

    mQuarterPlanClosing = False
    If mQuarterPlanUiBusy Then Exit Sub

    mQuarterPlanActionRunning = True
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    Set ws = GetQuarterPlanWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Quarter plan worksheet not found by tblPlanActive."
    Set sourceTbl = ws.ListObjects("tblPlanBacklog")
    Set targetTbl = ws.ListObjects("tblPlanActive")

    movedCount = MoveAllQuarterPlanTasksFast(sourceTbl, targetTbl)
    RefreshQuarterPlanTable sourceTbl
    RefreshQuarterPlanTable targetTbl
    SyncQuarterPlanTableActionButtons sourceTbl
    SyncQuarterPlanTableActionButtons targetTbl

    If movedCount = 0 Then
        MsgBox "Не перенесено ни одной задачи: бэклог пуст или в активном плане нет свободных строк.", vbInformation
    End If

SafeExit:
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    mQuarterPlanActionRunning = False
    Exit Sub

Fail:
    MsgBox "Ошибка RunQuarterPlanAddAllBacklogToPlan: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
    Resume SafeExit
End Sub
