Private Sub Workbook_Open()
    Dim wsPlan As Worksheet
    Dim wsEstimate As Worksheet
    Dim oldEnableEvents As Boolean
    Dim oldScreenUpdating As Boolean

    On Error GoTo SafeExit
    oldEnableEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    mQuarterPlanClosing = False
    mQuarterPlanUiBusy = True
    Set wsPlan = GetQuarterPlanWorksheet()
    If Not wsPlan Is Nothing Then SetupQuarterPlanActionUi wsPlan
    Set wsEstimate = GetTaskEstimatesWorksheet()
    If Not wsEstimate Is Nothing Then
        RefreshTaskEstimateTable wsEstimate.ListObjects("tblTaskEstimates")
        AddTaskEstimateTopButtons wsEstimate
    End If
SafeExit:
    mQuarterPlanUiBusy = False
    Application.ScreenUpdating = oldScreenUpdating
    Application.EnableEvents = oldEnableEvents
End Sub

Private Sub Workbook_BeforeClose(Cancel As Boolean)
    mQuarterPlanClosing = False
    mQuarterPlanUiBusy = False
    mQuarterPlanActionRunning = False
End Sub

Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    Dim oldEnableEvents As Boolean

    On Error GoTo Fail

    oldEnableEvents = Application.EnableEvents
    Application.EnableEvents = False

    If WorksheetHasTable(Sh, "tblHolidays") Then
        If Not Intersect(Target, Sh.Range("B4:B6")) Is Nothing Then
            RefreshQuarterHolidays Sh
        End If
    End If

    If WorksheetHasTable(Sh, "tblTeamComposition") And WorksheetHasTable(Sh, "tblTeamMembers") Then
        If IsTeamCompositionCountTarget(Sh, Target) Then
            SyncTeamMembersFromComposition Sh
        End If
    End If

    If WorksheetHasTable(Sh, "tblTaskEstimates") Then
        MarkTaskEstimateRefreshForTarget Sh, Target
    End If

SafeExit:
    Application.EnableEvents = oldEnableEvents
    Exit Sub

Fail:
    MsgBox "Ошибка Workbook_SheetChange: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Sheet=" & Sh.Name & ", Target=" & Target.Address(False, False), vbExclamation
    Resume SafeExit
End Sub

Private Function TryRunTaskEstimateCellAction(ByVal Sh As Object, ByVal Target As Range) As Boolean
    On Error GoTo Fail

    If mQuarterPlanUiBusy Or mQuarterPlanActionRunning Then Exit Function
    If Target.CountLarge <> 1 Then Exit Function
    If Not WorksheetHasTable(Sh, "tblTaskEstimates") Then Exit Function
    If Not IsTaskEstimateActionCell(Sh, Target) Then Exit Function

    mQuarterPlanActionRunning = True
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    If HandleTaskEstimateAction(Sh, Target) Then
        RefreshTaskEstimateActionGrid
        TryRunTaskEstimateCellAction = True
    End If

SafeExit:
    Application.ScreenUpdating = True
    mQuarterPlanActionRunning = False
    Application.EnableEvents = True
    Exit Function

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "TryRunTaskEstimateCellAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Sheet: " & Sh.Name & vbCrLf & _
        "Cell: " & Target.Address(False, False), vbExclamation
    Resume SafeExit
End Function

Private Function TryRunQuarterPlanCellAction(ByVal Sh As Object, ByVal Target As Range) As Boolean
    On Error GoTo SafeExit

    Dim actionTarget As Range

    If mQuarterPlanUiBusy Or mQuarterPlanActionRunning Then Exit Function
    If Not WorksheetHasTable(Sh, "tblPlanActive") Then Exit Function

    Set actionTarget = Target
    If actionTarget.MergeCells Then Set actionTarget = actionTarget.MergeArea.Cells(1, 1)
    If actionTarget.CountLarge <> 1 Then Exit Function

    If actionTarget.Address(False, False) = "A54" And Trim$(CStr(actionTarget.Value)) = QuarterPlanTextAddAllToPlan() Then
        mQuarterPlanActionRunning = True
        Application.EnableEvents = False
        Application.ScreenUpdating = False
        RunQuarterPlanAddAllBacklogToPlan
        TryRunQuarterPlanCellAction = True
        GoTo SafeExit
    End If

    If actionTarget.Address(False, False) = "Q2" Then
        mQuarterPlanActionRunning = True
        Application.EnableEvents = False
        ButtonExportQuarterPlanBacklog
        TryRunQuarterPlanCellAction = True
        GoTo SafeExit
    End If

    If IsQuarterPlanActionCell(Sh, actionTarget) Then
        mQuarterPlanActionRunning = True
        Application.ScreenUpdating = False
        If HandleQuarterPlanAction(Sh, actionTarget) Then
            TryRunQuarterPlanCellAction = True
        End If
    End If

SafeExit:
    Application.ScreenUpdating = True
    mQuarterPlanActionRunning = False
    Application.EnableEvents = True
End Function

Public Sub HandleQuarterPlanShapeAction(ByVal shapeName As String)
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim shp As Shape
    Dim actionCell As Range
    Dim normalizedShapeName As String
    Dim actionHandled As Boolean

    If mQuarterPlanUiBusy Or mQuarterPlanActionRunning Then Exit Sub

    Set ws = GetQuarterPlanWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Quarter plan worksheet not found by tblPlanActive."
    normalizedShapeName = NormalizeShapeCallerName(shapeName)
    Set shp = ws.Shapes(normalizedShapeName)
    Set actionCell = shp.TopLeftCell

    mQuarterPlanActionRunning = True
    Application.EnableEvents = False

    If actionCell.MergeCells Then Set actionCell = actionCell.MergeArea.Cells(1, 1)

    If actionCell.Address(False, False) = "A54" And Trim$(CStr(actionCell.Value)) = QuarterPlanTextAddAllToPlan() Then
        Application.ScreenUpdating = False
        RunQuarterPlanAddAllBacklogToPlan
    ElseIf IsQuarterPlanActionCell(ws, actionCell) Then
        Application.ScreenUpdating = False
        actionHandled = HandleQuarterPlanAction(ws, actionCell)
    End If

SafeExit:
    Application.ScreenUpdating = True
    mQuarterPlanActionRunning = False
    Application.EnableEvents = True
    Exit Sub

Fail:
    If Left$(normalizedShapeName, Len(QUARTER_PLAN_CELL_ACTION_LEGACY_PREFIX)) = QUARTER_PLAN_CELL_ACTION_LEGACY_PREFIX Then
        On Error Resume Next
        RebuildQuarterPlanActionUi
        On Error GoTo 0
        MsgBox "Кнопки действий были созданы старой версией с длинными именами и Excel вернул обрезанное имя." & vbCrLf & _
            "Интерфейс кнопок пересоздан. Нажмите действие еще раз." & vbCrLf & _
            "Caller: " & shapeName & vbCrLf & _
            "Shape: " & normalizedShapeName, vbInformation
        Resume SafeExit
    End If

    MsgBox "Ошибка HandleQuarterPlanShapeAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Caller: " & shapeName & vbCrLf & _
        "Shape: " & normalizedShapeName, vbExclamation
    Resume SafeExit
End Sub

    On Error GoTo Fail

    Dim ws As Worksheet

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "HandleTaskEstimateShapeAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Caller: " & shapeName & vbCrLf & _
        "Shape: " & normalizedShapeName, vbExclamation
    Resume SafeExit
End Sub

Private Function NormalizeShapeCallerName(ByVal callerName As String) As String
    Dim result As String
    Dim bangPos As Long

    result = Trim$(callerName)
    result = Replace(result, "'", "")

    bangPos = InStrRev(result, "!")
    If bangPos > 0 Then
        result = Mid$(result, bangPos + 1)
    End If

    NormalizeShapeCallerName = result
End Function

Private Function FindWorksheetByTableName(ByVal tableName As String) As Worksheet
    Dim ws As Worksheet
    Dim tbl As ListObject

    For Each ws In ThisWorkbook.Worksheets
        Set tbl = Nothing
        On Error Resume Next
        Set tbl = ws.ListObjects(tableName)
        On Error GoTo 0
        If Not tbl Is Nothing Then
            Set FindWorksheetByTableName = ws
            Exit Function
        End If
    Next ws
End Function

Private Function WorksheetHasTable(ByVal ws As Worksheet, ByVal tableName As String) As Boolean
    Dim tbl As ListObject

    If ws Is Nothing Then Exit Function
    On Error Resume Next
    Set tbl = ws.ListObjects(tableName)
    On Error GoTo 0
    WorksheetHasTable = Not tbl Is Nothing
End Function

Private Function IsTeamCompositionCountTarget(ByVal ws As Worksheet, ByVal Target As Range) As Boolean
    Dim tbl As ListObject
    Dim countRange As Range

    If ws Is Nothing Then Exit Function
    If Target Is Nothing Then Exit Function

    Set tbl = ws.ListObjects("tblTeamComposition")
    If tbl Is Nothing Then Exit Function
    If tbl.DataBodyRange Is Nothing Then Exit Function

    Set countRange = tbl.ListColumns(2).DataBodyRange
    IsTeamCompositionCountTarget = Not Intersect(Target, countRange) Is Nothing
End Function

Private Sub SyncTeamMembersFromComposition(ByVal ws As Worksheet)
    Dim compositionTbl As ListObject
    Dim membersTbl As ListObject
    Dim oldValues As Variant
    Dim newValues() As Variant
    Dim usedRows() As Boolean
    Dim memberRows As Long
    Dim memberCols As Long
    Dim roleRow As Long
    Dim roleText As String
    Dim roleCount As Long
    Dim roleMemberIndex As Long
    Dim outputRow As Long
    Dim matchedRow As Long
    Dim colIndex As Long
    Dim syncStep As String

    On Error GoTo Fail

    syncStep = "resolve tables"
    Set compositionTbl = ws.ListObjects("tblTeamComposition")
    Set membersTbl = ws.ListObjects("tblTeamMembers")
    If compositionTbl Is Nothing Then Err.Raise 9, , "tblTeamComposition not found."
    If membersTbl Is Nothing Then Err.Raise 9, , "tblTeamMembers not found."
    If compositionTbl.DataBodyRange Is Nothing Then Exit Sub
    If membersTbl.DataBodyRange Is Nothing Then Exit Sub

    memberRows = membersTbl.DataBodyRange.Rows.Count
    memberCols = membersTbl.DataBodyRange.Columns.Count
    oldValues = membersTbl.DataBodyRange.Value
    ReDim newValues(1 To memberRows, 1 To memberCols)
    ReDim usedRows(1 To memberRows)

    outputRow = 0
    For roleRow = 1 To compositionTbl.DataBodyRange.Rows.Count
        roleText = Trim$(CStr(compositionTbl.DataBodyRange.Cells(roleRow, 1).Value))
        roleCount = TeamMemberCountFromValue(compositionTbl.DataBodyRange.Cells(roleRow, 2).Value)
        If roleText <> "" And roleCount > 0 Then
            For roleMemberIndex = 1 To roleCount
                If outputRow >= memberRows Then Exit For
                outputRow = outputRow + 1
                newValues(outputRow, TEAM_MEMBER_COL_ROLE) = roleText
                matchedRow = FindReusableTeamMemberRow(oldValues, usedRows, roleText)
                If matchedRow > 0 Then
                    For colIndex = TEAM_MEMBER_COL_NAME To memberCols
                        If Not IsTeamMemberCalculatedColumn(colIndex) Then
                            newValues(outputRow, colIndex) = oldValues(matchedRow, colIndex)
                        End If
                    Next colIndex
                    If Trim$(CStr(newValues(outputRow, TEAM_MEMBER_COL_NAME))) = "" Then
                        newValues(outputRow, TEAM_MEMBER_COL_NAME) = roleText & " " & CStr(roleMemberIndex)
                    End If
                    If Trim$(CStr(newValues(outputRow, TEAM_MEMBER_COL_ALLOCATION))) = "" Then
                        newValues(outputRow, TEAM_MEMBER_COL_ALLOCATION) = 1
                    End If
                Else
                    newValues(outputRow, TEAM_MEMBER_COL_NAME) = roleText & " " & CStr(roleMemberIndex)
                    newValues(outputRow, TEAM_MEMBER_COL_ALLOCATION) = 1
                End If
            Next roleMemberIndex
        End If
    Next roleRow

    syncStep = "write editable member values"
    WriteTeamMemberEditableValues membersTbl, newValues, memberRows
    syncStep = "format allocation"
    membersTbl.ListColumns(TEAM_MEMBER_COL_ALLOCATION).DataBodyRange.NumberFormat = "0%"
    syncStep = "format vacation dates"
    ApplyDateFormat ws.Range("D24:E43")
    ApplyDateFormat ws.Range("G24:H43")
    ApplyDateFormat ws.Range("J24:K43")
    Exit Sub

Fail:
    Err.Raise Err.Number, "SyncTeamMembersFromComposition", "Step=" & syncStep & ": " & Err.Description
End Sub

Private Function IsTeamMemberCalculatedColumn(ByVal columnIndex As Long) As Boolean
    IsTeamMemberCalculatedColumn = (columnIndex = TEAM_MEMBER_COL_VACATION1_DAYS Or _
        columnIndex = TEAM_MEMBER_COL_VACATION2_DAYS Or _
        columnIndex = TEAM_MEMBER_COL_VACATION3_DAYS)
End Function

Private Sub WriteTeamMemberEditableValues(ByVal membersTbl As ListObject, ByRef values() As Variant, ByVal rowCount As Long)
    Dim coreValues() As Variant
    Dim vacation1Values() As Variant
    Dim vacation2Values() As Variant
    Dim vacation3Values() As Variant
    Dim rowIndex As Long

    ReDim coreValues(1 To rowCount, 1 To 3)
    ReDim vacation1Values(1 To rowCount, 1 To 2)
    ReDim vacation2Values(1 To rowCount, 1 To 2)
    ReDim vacation3Values(1 To rowCount, 1 To 2)

    For rowIndex = 1 To rowCount
        coreValues(rowIndex, 1) = values(rowIndex, TEAM_MEMBER_COL_ROLE)
        coreValues(rowIndex, 2) = values(rowIndex, TEAM_MEMBER_COL_NAME)
        coreValues(rowIndex, 3) = values(rowIndex, TEAM_MEMBER_COL_ALLOCATION)
        vacation1Values(rowIndex, 1) = values(rowIndex, TEAM_MEMBER_COL_VACATION1_START)
        vacation1Values(rowIndex, 2) = values(rowIndex, TEAM_MEMBER_COL_VACATION1_END)
        vacation2Values(rowIndex, 1) = values(rowIndex, TEAM_MEMBER_COL_VACATION2_START)
        vacation2Values(rowIndex, 2) = values(rowIndex, TEAM_MEMBER_COL_VACATION2_END)
        vacation3Values(rowIndex, 1) = values(rowIndex, TEAM_MEMBER_COL_VACATION3_START)
        vacation3Values(rowIndex, 2) = values(rowIndex, TEAM_MEMBER_COL_VACATION3_END)
    Next rowIndex

    membersTbl.DataBodyRange.Cells(1, TEAM_MEMBER_COL_ROLE).Resize(rowCount, 3).Value2 = coreValues
    membersTbl.DataBodyRange.Cells(1, TEAM_MEMBER_COL_VACATION1_START).Resize(rowCount, 2).Value2 = vacation1Values
    membersTbl.DataBodyRange.Cells(1, TEAM_MEMBER_COL_VACATION2_START).Resize(rowCount, 2).Value2 = vacation2Values
    membersTbl.DataBodyRange.Cells(1, TEAM_MEMBER_COL_VACATION3_START).Resize(rowCount, 2).Value2 = vacation3Values
End Sub

Private Function TeamMemberCountFromValue(ByVal rawValue As Variant) As Long
    If IsNumeric(rawValue) Then
        TeamMemberCountFromValue = CLng(Fix(CDbl(rawValue)))
        If TeamMemberCountFromValue < 0 Then TeamMemberCountFromValue = 0
        If TeamMemberCountFromValue > TEAM_MEMBER_LIMIT Then TeamMemberCountFromValue = TEAM_MEMBER_LIMIT
    End If
End Function

Private Function FindReusableTeamMemberRow(ByRef oldValues As Variant, ByRef usedRows() As Boolean, ByVal roleText As String) As Long
    Dim rowIndex As Long

    For rowIndex = LBound(oldValues, 1) To UBound(oldValues, 1)
        If Not usedRows(rowIndex) Then
            If Trim$(CStr(oldValues(rowIndex, TEAM_MEMBER_COL_ROLE))) = roleText Then
                usedRows(rowIndex) = True
                FindReusableTeamMemberRow = rowIndex
                Exit Function
            End If
        End If
    Next rowIndex
End Function

Private Sub RefreshQuarterHolidays(ByVal ws As Worksheet)
    Dim startDate As Variant
    Dim endDate As Variant
    Dim workingDays As Variant
    Dim calendarWeekdays As Long
    Dim holidayCount As Long
    Dim tbl As ListObject
    Dim oldLastRow As Long
    Dim clearLastRow As Long
    Dim targetLastRow As Long
    Dim i As Long
    Dim yesText As String

    startDate = ws.Range("B4").Value
    endDate = ws.Range("B5").Value
    workingDays = ws.Range("B6").Value

    Set tbl = ws.ListObjects("tblHolidays")
    ws.Range("B7:B8").NumberFormat = "0"

    oldLastRow = tbl.Range.Row + tbl.Range.Rows.Count - 1
    clearLastRow = Application.WorksheetFunction.Max(oldLastRow, 33)
    ws.Range("A14:D" & clearLastRow).ClearContents

    If Not IsDate(startDate) Or Not IsDate(endDate) Or Not IsNumeric(workingDays) Then
        ws.Range("B7:B8").ClearContents
        tbl.Resize ws.Range("A13:D14")
        ApplyHolidayTableFormatting tbl
        Exit Sub
    End If

    calendarWeekdays = Application.WorksheetFunction.NetworkDays(CDate(startDate), CDate(endDate))
    holidayCount = Application.WorksheetFunction.Max(0, calendarWeekdays - CLng(workingDays))

    ws.Range("B7").Value = calendarWeekdays
    ws.Range("B8").Value = holidayCount

    If holidayCount = 0 Then
        tbl.Resize ws.Range("A13:D14")
        ApplyHolidayTableFormatting tbl
        Exit Sub
    End If

    targetLastRow = 13 + holidayCount
    tbl.Resize ws.Range("A13:D" & targetLastRow)
    ApplyHolidayTableFormatting tbl

    yesText = ChrW$(&H414) & ChrW$(&H430)

    For i = 1 To holidayCount
        tbl.DataBodyRange.Cells(i, 1).Value = Application.WorksheetFunction.WorkDay(CDate(startDate) - 1, i)
        tbl.DataBodyRange.Cells(i, 2).ClearContents
        tbl.DataBodyRange.Cells(i, 3).ClearContents
        tbl.DataBodyRange.Cells(i, 4).Value = yesText
    Next i

    ApplyDateFormat tbl.ListColumns(1).DataBodyRange
End Sub

Private Function RestoreTaskEstimateTemplate(ByVal ws As Worksheet, ByRef tbl As ListObject) As Boolean
    Dim targetRange As Range

    On Error GoTo Fail

    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet is not available."
    If tbl Is Nothing Then Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl Is Nothing Then Err.Raise 9, , "tblTaskEstimates not found."

    If tbl.Range.Row <> TASK_ESTIMATE_TEMPLATE_FIRST_ROW Or tbl.Range.Column <> EST_COL_PRIORITY + 2 Then
        Err.Raise 5, , "tblTaskEstimates must start at C3 for template recovery."
    End If

    Set targetRange = ws.Range(TASK_ESTIMATE_TEMPLATE_RANGE)
    If tbl.Range.Address(False, False) <> targetRange.Address(False, False) Then
        tbl.Resize targetRange
        RestoreTaskEstimateTemplate = True
        Set tbl = ws.ListObjects("tblTaskEstimates")
    End If

    RestoreTaskEstimateTemplateHeader ws
    RestoreTaskEstimateTemplateStyles ws, tbl
    RestoreTaskEstimateTemplateValidation ws
    RestoreTaskEstimateTemplateStats ws
    Exit Function

Fail:
    Err.Raise Err.Number, "RestoreTaskEstimateTemplate", Err.Description
End Function

Private Function TextFromCodes(ParamArray codepoints() As Variant) As String
    Dim i As Long
    Dim result As String

    For i = LBound(codepoints) To UBound(codepoints)

Private Function SanitizeFileName(ByVal fileName As String) As String
    Dim invalidChars As Variant
    Dim item As Variant

    invalidChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For Each item In invalidChars
        fileName = Replace(fileName, CStr(item), "_")
    Next item
    SanitizeFileName = Trim$(fileName)
End Function

Private Function ExportTaskEstimatesToXlsxFile(ByVal exportPath As String, ByVal showMessages As Boolean, ByVal allowOverwrite As Boolean) As Long
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim exportWb As Workbook
    Dim exportWs As Worksheet
    Dim lastRow As Long
    Dim rowIndex As Long
    Dim outRow As Long
    Dim colIndex As Long
    Dim normalizedPath As String
    Dim oldEnableEvents As Boolean
    Dim oldScreenUpdating As Boolean
    Dim oldDisplayAlerts As Boolean

    oldEnableEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating
    oldDisplayAlerts = Application.DisplayAlerts

    normalizedPath = EnsureXlsxExtension(Trim$(exportPath))
    If normalizedPath = "" Then Err.Raise 5, , "Export path is empty."
    If Dir$(normalizedPath) <> "" And Not allowOverwrite Then
        If MsgBox(TaskEstimateTextFileExistsPrompt() & vbCrLf & normalizedPath, vbYesNo + vbQuestion, TaskEstimateTextExportDialogTitle()) <> vbYes Then Exit Function
    End If

    Set ws = GetTaskEstimatesWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl.DataBodyRange Is Nothing Then Err.Raise 9, , "tblTaskEstimates has no data body range."

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    RefreshTaskEstimateTable tbl
    lastRow = LastTaskEstimateUsedRow(tbl)
    If lastRow = 0 Then
        If showMessages Then
        MsgBox TaskEstimateTextNoTasksForExport(), vbInformation
            GoTo SafeExit
        Else
            Err.Raise 5, , "No task estimate rows to export."
        End If
    End If

    Set exportWb = Application.Workbooks.Add(xlWBATWorksheet)
    Set exportWs = exportWb.Worksheets(1)
    exportWs.Name = TaskEstimateTextSheetName()
    exportWs.Cells.Font.Name = DESIGN_FONT_NAME
    exportWs.Cells.Font.Size = DESIGN_FONT_SIZE

    For colIndex = EST_COL_PRIORITY To EST_COL_LAST
        exportWs.Cells(1, colIndex - EST_COL_PRIORITY + 1).Value = tbl.HeaderRowRange.Cells(1, colIndex).Value
        exportWs.Columns(colIndex - EST_COL_PRIORITY + 1).ColumnWidth = tbl.Parent.Columns(tbl.Range.Column + colIndex - 1).ColumnWidth
    Next colIndex

    outRow = 2
    For rowIndex = 1 To lastRow
        If TaskEstimateRowHasTask(tbl, rowIndex) Then
            For colIndex = EST_COL_PRIORITY To EST_COL_LAST
                exportWs.Cells(outRow, colIndex - EST_COL_PRIORITY + 1).Value = tbl.DataBodyRange.Cells(rowIndex, colIndex).Value
                exportWs.Cells(outRow, colIndex - EST_COL_PRIORITY + 1).NumberFormat = tbl.DataBodyRange.Cells(rowIndex, colIndex).NumberFormat
            Next colIndex
            exportWs.Cells(outRow, EST_COL_DESCRIPTION - EST_COL_PRIORITY + 1).IndentLevel = tbl.DataBodyRange.Cells(rowIndex, EST_COL_DESCRIPTION).IndentLevel
            outRow = outRow + 1
        End If
    Next rowIndex

    With exportWs.Range(exportWs.Cells(1, 1), exportWs.Cells(1, EST_COL_LAST - EST_COL_PRIORITY + 1))
        .Font.Name = DESIGN_FONT_NAME
        .Font.Size = DESIGN_FONT_SIZE
        .Font.Bold = True
        .Interior.Color = RGB(217, 234, 247)
        .HorizontalAlignment = xlCenter
    End With
    exportWs.Range(exportWs.Cells(1, 1), exportWs.Cells(outRow - 1, EST_COL_LAST - EST_COL_PRIORITY + 1)).Borders.LineStyle = xlContinuous
    exportWs.Range(exportWs.Cells(2, EST_COL_AN - EST_COL_PRIORITY + 1), exportWs.Cells(outRow - 1, EST_COL_QA - EST_COL_PRIORITY + 1)).NumberFormat = "0.00"
    exportWs.Rows(1).AutoFilter

    Application.DisplayAlerts = False
    exportWb.SaveAs Filename:=normalizedPath, FileFormat:=xlOpenXMLWorkbook
    Application.DisplayAlerts = oldDisplayAlerts

    ExportTaskEstimatesToXlsxFile = outRow - 2
    If showMessages Then MsgBox TaskEstimateTextExportedTasks() & CStr(ExportTaskEstimatesToXlsxFile) & vbCrLf & normalizedPath, vbInformation

SafeExit:
    On Error Resume Next
    If Not exportWb Is Nothing Then exportWb.Close SaveChanges:=False
    Application.DisplayAlerts = oldDisplayAlerts
    Application.ScreenUpdating = oldScreenUpdating
    Application.EnableEvents = oldEnableEvents
    On Error GoTo 0
    Exit Function

Fail:
    If showMessages Then
        MsgBox TaskEstimateTextErrorPrefix() & "ExportTaskEstimatesToXlsxFile: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
            "Path: " & exportPath, vbExclamation
    Else
        Err.Raise Err.Number, "ExportTaskEstimatesToXlsxFile", Err.Description
    End If
    Resume SafeExit
End Function

Private Function ImportTaskEstimatesXlsxFile(ByVal xlsxPath As String, ByVal showSuccess As Boolean, Optional ByVal showErrors As Boolean = True) As Long
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim importWb As Workbook
    Dim importWs As Worksheet
    Dim importedValues() As Variant
    Dim importedLevels() As Long
    Dim rowIndex As Long
    Dim colIndex As Long
    Dim lastImportRow As Long
    Dim importedCount As Long
    Dim maxRows As Long
    Dim usedRows As Long
    Dim targetStartRow As Long
    Dim targetRow As Long
    Dim taskLevel As Long
    Dim previousLevel As Long
    Dim parentIds(0 To EST_MAX_DECOMPOSITION_LEVEL) As String
    Dim currentId As String
    Dim oldEnableEvents As Boolean
    Dim oldScreenUpdating As Boolean
    Dim oldDisplayAlerts As Boolean

    oldEnableEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating
    oldDisplayAlerts = Application.DisplayAlerts

    If Trim$(xlsxPath) = "" Then Err.Raise 5, , "XLSX path is empty."
    If Dir$(xlsxPath) = "" Then Err.Raise 53, , "XLSX file not found."

    Set ws = GetTaskEstimatesWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl.DataBodyRange Is Nothing Then Err.Raise 9, , "tblTaskEstimates has no data body range."

    Application.EnableEvents = False
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    Set importWb = Application.Workbooks.Open(Filename:=xlsxPath, UpdateLinks:=0, ReadOnly:=True)
    Set importWs = importWb.Worksheets(1)
    ValidateTaskEstimateXlsxHeaders importWs

    lastImportRow = LastTaskEstimateImportXlsxRow(importWs)
    If lastImportRow < 2 Then Err.Raise 5, , "XLSX file has no task rows to import."

    ReDim importedValues(1 To lastImportRow - 1, 1 To EST_COL_COMMENT)
    ReDim importedLevels(1 To lastImportRow - 1)
    previousLevel = 0

    For rowIndex = 2 To lastImportRow
        If Not TaskEstimateImportXlsxRowHasData(importWs, rowIndex) Then
            Err.Raise 5, , "XLSX contains an empty task row before the end of imported data. Row: " & CStr(rowIndex)
        End If

        taskLevel = CLng(importWs.Cells(rowIndex, EST_COL_DESCRIPTION).IndentLevel)
        If taskLevel < 0 Or taskLevel > EST_MAX_DECOMPOSITION_LEVEL Then
            Err.Raise 5, , "Unsupported task indentation level at row " & CStr(rowIndex) & ": " & CStr(taskLevel)
        End If
        If rowIndex = 2 And taskLevel <> 0 Then
            Err.Raise 5, , "First imported task must be a root task."
        End If
        If taskLevel > previousLevel + 1 Then
            Err.Raise 5, , "XLSX task hierarchy skips a level at row " & CStr(rowIndex) & "."
        End If

        importedCount = importedCount + 1
        importedLevels(importedCount) = taskLevel
        For colIndex = EST_COL_PRIORITY To EST_COL_COMMENT
            importedValues(importedCount, colIndex) = importWs.Cells(rowIndex, colIndex).Value
        Next colIndex
        previousLevel = taskLevel
    Next rowIndex

    maxRows = tbl.DataBodyRange.Rows.Count
    usedRows = LastTaskEstimateUsedRow(tbl)
    If importedCount > maxRows - usedRows Then
        Err.Raise 5, , "XLSX contains " & CStr(importedCount) & " task rows, but only " & CStr(maxRows - usedRows) & " rows are available for append import."
    End If

    targetStartRow = usedRows + 1
    For rowIndex = 1 To importedCount
        targetRow = targetStartRow + rowIndex - 1
        taskLevel = importedLevels(rowIndex)
        ClearTaskEstimateRow tbl, targetRow
        For colIndex = EST_COL_PRIORITY To EST_COL_COMMENT
            tbl.DataBodyRange.Cells(targetRow, colIndex).Value = importedValues(rowIndex, colIndex)
        Next colIndex

        currentId = NewTaskEstimateId()
        TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_TASK_ID).Value = currentId
        If taskLevel = 0 Then
            TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_PARENT_TASK_ID).ClearContents
        Else
            If parentIds(taskLevel - 1) = "" Then Err.Raise 5, , "Parent task is missing for imported row " & CStr(rowIndex) & "."
            TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_PARENT_TASK_ID).Value = parentIds(taskLevel - 1)
        End If
        TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_LEVEL).Value = taskLevel
        tbl.DataBodyRange.Cells(targetRow, EST_COL_DESCRIPTION).IndentLevel = taskLevel
        parentIds(taskLevel) = currentId
        For colIndex = taskLevel + 1 To EST_MAX_DECOMPOSITION_LEVEL
            parentIds(colIndex) = ""
        Next colIndex
    Next rowIndex

    MarkTaskEstimateRefreshFrom targetStartRow
    RefreshTaskEstimateTable tbl
    AddTaskEstimateTopButtons ws
    SyncTaskEstimateActionButtons ws

    ImportTaskEstimatesXlsxFile = importedCount
    If showSuccess Then MsgBox TaskEstimateTextImportedTasks() & CStr(importedCount), vbInformation

SafeExit:
    On Error Resume Next
    If Not importWb Is Nothing Then importWb.Close SaveChanges:=False
    Application.DisplayAlerts = oldDisplayAlerts
    Application.ScreenUpdating = oldScreenUpdating
    Application.EnableEvents = oldEnableEvents
    On Error GoTo 0
    Exit Function

Fail:
    If showErrors Then
        MsgBox TaskEstimateTextErrorPrefix() & "ImportTaskEstimatesXlsxFile: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
            "Path: " & xlsxPath, vbExclamation
    Else
        Err.Raise Err.Number, "ImportTaskEstimatesXlsxFile", Err.Description
    End If
    Resume SafeExit
End Function

Private Sub ValidateTaskEstimateXlsxHeaders(ByVal importWs As Worksheet)
    Dim expectedHeaders As Variant
    Dim index As Long
    Dim actualValue As String

    expectedHeaders = TaskEstimateVisibleHeaders()
    For index = LBound(expectedHeaders) To UBound(expectedHeaders)
        actualValue = Trim$(CStr(importWs.Cells(1, index + 1).Value))
        If actualValue <> CStr(expectedHeaders(index)) Then
            Err.Raise 5, , "Unexpected XLSX header in column " & CStr(index + 1) & ": " & actualValue
        End If
    Next index
End Sub

Private Function LastTaskEstimateImportXlsxRow(ByVal importWs As Worksheet) As Long
    Dim rowIndex As Long
    Dim colIndex As Long
    Dim candidateRow As Long
    Dim lastUsedRow As Long

    For colIndex = EST_COL_PRIORITY To EST_COL_COMMENT
        candidateRow = importWs.Cells(importWs.Rows.Count, colIndex).End(xlUp).Row
        If candidateRow > lastUsedRow Then lastUsedRow = candidateRow
    Next colIndex
    For rowIndex = lastUsedRow To 2 Step -1
        If TaskEstimateImportXlsxRowHasData(importWs, rowIndex) Then
            LastTaskEstimateImportXlsxRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function TaskEstimateImportXlsxRowHasData(ByVal importWs As Worksheet, ByVal rowIndex As Long) As Boolean
    Dim colIndex As Long

    For colIndex = EST_COL_PRIORITY To EST_COL_COMMENT
        If Trim$(CStr(importWs.Cells(rowIndex, colIndex).Value)) <> "" Then
            TaskEstimateImportXlsxRowHasData = True
            Exit Function
        End If
    Next colIndex
End Function

Private Sub PrepareExpressEstimateRows(ByVal exportWs As Worksheet, ByVal tbl As ListObject, ByVal firstSourceRow As Long, ByVal lastSourceRow As Long)
    Dim sourceRow As Long
    Dim outRow As Long
    Dim rowCount As Long

    rowCount = lastSourceRow - firstSourceRow + 1
    If rowCount < 1 Then rowCount = 1

    If rowCount > 7 Then
        For outRow = 24 To 17 + rowCount - 1
            exportWs.Rows(outRow).Insert
        Next outRow
    End If

    exportWs.Range("A17:L" & CStr(16 + rowCount)).ClearContents
    If rowCount < 7 Then exportWs.Range("A" & CStr(17 + rowCount) & ":L23").ClearContents

    outRow = 17
    For sourceRow = firstSourceRow To lastSourceRow
        If TaskEstimateRowHasTask(tbl, sourceRow) Then
            FormatExpressEstimateRow exportWs, outRow, TaskEstimateRowLevel(tbl, sourceRow)
            outRow = outRow + 1
        End If
    Next sourceRow
End Sub

Private Sub FormatExpressEstimateRow(ByVal exportWs As Worksheet, ByVal targetRow As Long, ByVal taskLevel As Long)
    Dim templateRow As Long

    If taskLevel <= 0 Then
        templateRow = 17
    ElseIf taskLevel = 1 Then
        templateRow = 18
    Else
        templateRow = 19
    End If

    exportWs.Range("A" & CStr(templateRow) & ":L" & CStr(templateRow)).Copy
    exportWs.Range("A" & CStr(targetRow) & ":L" & CStr(targetRow)).PasteSpecial xlPasteFormats
    Application.CutCopyMode = False
    exportWs.Range("A" & CStr(targetRow) & ":D" & CStr(targetRow)).UnMerge

    Select Case taskLevel
        Case 0
            exportWs.Range("A" & CStr(targetRow) & ":D" & CStr(targetRow)).Merge
        Case 1
            exportWs.Range("B" & CStr(targetRow) & ":D" & CStr(targetRow)).Merge
        Case Else
            exportWs.Range("C" & CStr(targetRow) & ":D" & CStr(targetRow)).Merge
    End Select
    exportWs.Range("A" & CStr(targetRow) & ":L" & CStr(targetRow)).ClearContents
End Sub

Private Sub FillExpressEstimateRow(ByVal exportWs As Worksheet, ByVal targetRow As Long, ByVal tbl As ListObject, ByVal sourceRow As Long)
    Dim taskLevel As Long

    taskLevel = TaskEstimateRowLevel(tbl, sourceRow)
    Select Case taskLevel
        Case 0
            exportWs.Range("A" & CStr(targetRow)).Value = tbl.DataBodyRange.Cells(sourceRow, EST_COL_DESCRIPTION).Value
        Case 1
            exportWs.Range("B" & CStr(targetRow)).Value = tbl.DataBodyRange.Cells(sourceRow, EST_COL_DESCRIPTION).Value
        Case Else
            exportWs.Range("C" & CStr(targetRow)).Value = tbl.DataBodyRange.Cells(sourceRow, EST_COL_DESCRIPTION).Value
    End Select

    WriteExpressEstimateValue exportWs.Range("G" & CStr(targetRow)), tbl.DataBodyRange.Cells(sourceRow, EST_COL_AN).Value
    exportWs.Range("H" & CStr(targetRow)).ClearContents
    WriteExpressEstimateValue exportWs.Range("I" & CStr(targetRow)), tbl.DataBodyRange.Cells(sourceRow, EST_COL_BE).Value
    WriteExpressEstimateValue exportWs.Range("J" & CStr(targetRow)), tbl.DataBodyRange.Cells(sourceRow, EST_COL_FE).Value
    WriteExpressEstimateValue exportWs.Range("K" & CStr(targetRow)), tbl.DataBodyRange.Cells(sourceRow, EST_COL_QA).Value
End Sub

Private Sub WriteExpressEstimateValue(ByVal targetCell As Range, ByVal sourceValue As Variant)
    If Trim$(CStr(sourceValue)) = "" Then
        targetCell.ClearContents

Private Function ImportTaskEstimatesCsvFile(ByVal csvPath As String, ByVal importMode As Long, ByVal showSuccess As Boolean, Optional ByVal showErrors As Boolean = True) As Long
    On Error GoTo Fail

    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim csvRows As Collection
    Dim headerRow As Variant
    Dim rowValues As Variant
    Dim ticketIndex As Long
    Dim subjectIndex As Long
    Dim rowIndex As Long
    Dim importedCount As Long
    Dim tickets() As String
    Dim descriptions() As String
    Dim ticketText As String
    Dim descriptionText As String
    Dim maxRows As Long
    Dim usedRows As Long
    Dim availableRows As Long
    Dim targetStartRow As Long
    Dim targetRow As Long
    Dim selectedMode As Long
    Dim oldEnableEvents As Boolean
    Dim oldScreenUpdating As Boolean

    oldEnableEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating

    If Trim$(csvPath) = "" Then Err.Raise 5, , "CSV path is empty."

    Set ws = GetTaskEstimatesWorksheet()
    If ws Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl.DataBodyRange Is Nothing Then Err.Raise 9, , "tblTaskEstimates has no data body range."

    Set csvRows = ParseCsvText(ReadUtf8TextFile(csvPath))
    If csvRows.Count < 1 Then Err.Raise 5, , "CSV file is empty."

    headerRow = csvRows.Item(1)
    ticketIndex = FindCsvHeaderIndex(headerRow, TaskEstimateCsvHeaderKey())
    subjectIndex = FindCsvHeaderIndex(headerRow, TaskEstimateCsvHeaderTheme())
    If ticketIndex = 0 Or subjectIndex = 0 Then
        Err.Raise 5, , "CSV must contain required headers: " & TaskEstimateCsvHeaderKey() & ", " & TaskEstimateCsvHeaderTheme() & "."
    End If

    For rowIndex = 2 To csvRows.Count
        rowValues = csvRows.Item(rowIndex)
        ticketText = CsvField(rowValues, ticketIndex)
        descriptionText = CsvField(rowValues, subjectIndex)
        If Trim$(ticketText) <> "" Or Trim$(descriptionText) <> "" Then
            importedCount = importedCount + 1
            If importedCount = 1 Then
                ReDim tickets(1 To 1)
                ReDim descriptions(1 To 1)
            Else
                ReDim Preserve tickets(1 To importedCount)
                ReDim Preserve descriptions(1 To importedCount)
            End If
            tickets(importedCount) = ticketText
            descriptions(importedCount) = descriptionText
        End If
    Next rowIndex

    If importedCount = 0 Then
        If showSuccess Then MsgBox TaskEstimateTextNoCsvRows(), vbInformation
        Exit Function
    End If

    maxRows = tbl.DataBodyRange.Rows.Count
    usedRows = LastTaskEstimateUsedRow(tbl)
    selectedMode = importMode
    If selectedMode = TASK_IMPORT_MODE_PROMPT Then
        selectedMode = PromptTaskEstimateImportMode(importedCount, maxRows, usedRows)
        If selectedMode = TASK_IMPORT_MODE_CANCEL Then Exit Function
    End If

    If selectedMode = TASK_IMPORT_MODE_REPLACE Then
        availableRows = maxRows
        targetStartRow = 1
    ElseIf selectedMode = TASK_IMPORT_MODE_APPEND Then
        availableRows = maxRows - usedRows
        targetStartRow = usedRows + 1
    Else
        Err.Raise 5, , "Unsupported import mode."
    End If

    If importedCount > availableRows Then
        Err.Raise 5, , "CSV contains " & CStr(importedCount) & " task rows, but only " & CStr(availableRows) & " rows are available for selected import mode."
    End If

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    If selectedMode = TASK_IMPORT_MODE_REPLACE Then
        For rowIndex = 1 To maxRows
            ClearTaskEstimateRow tbl, rowIndex
        Next rowIndex
        mTaskEstimateMaxActionRow = maxRows
        MarkTaskEstimateRefreshFrom 1
    End If

    For rowIndex = 1 To importedCount
        targetRow = targetStartRow + rowIndex - 1
        ClearTaskEstimateRow tbl, targetRow
        tbl.DataBodyRange.Cells(targetRow, EST_COL_DESCRIPTION).Value = descriptions(rowIndex)
        tbl.DataBodyRange.Cells(targetRow, EST_COL_TICKET).Value = tickets(rowIndex)
        TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_TASK_ID).Value = NewTaskEstimateId()
        TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_PARENT_TASK_ID).ClearContents
        TaskEstimateSheetCell(tbl, targetRow, EST_SHEET_COL_LEVEL).Value = 0
    Next rowIndex

    MarkTaskEstimateRefreshFrom targetStartRow
    RefreshTaskEstimateTable tbl
    AddTaskEstimateTopButtons ws
    SyncTaskEstimateActionButtons ws

    ImportTaskEstimatesCsvFile = importedCount
    If showSuccess Then
        MsgBox TaskEstimateTextImportedTasks() & CStr(importedCount), vbInformation
    End If

SafeExit:
    Application.ScreenUpdating = oldScreenUpdating
    Application.EnableEvents = oldEnableEvents
    Exit Function

Fail:
    If showErrors Then
        MsgBox TaskEstimateTextErrorPrefix() & "ImportTaskEstimatesCsvFile: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
            "Path: " & csvPath, vbExclamation
    Else
        Err.Raise Err.Number, "ImportTaskEstimatesCsvFile", Err.Description
    End If
    Resume SafeExit
End Function

Private Function ReadUtf8TextFile(ByVal filePath As String) As String
    Dim fileNumber As Integer
    Dim fileLength As Long
    Dim bytes() As Byte
    Dim index As Long
    Dim startIndex As Long
    Dim b1 As Long
    Dim b2 As Long
    Dim b3 As Long
    Dim b4 As Long
    Dim codePoint As Long
    Dim result As String

    If Dir$(filePath) = "" Then Err.Raise 53, , "CSV file not found."

    fileNumber = FreeFile
    Open filePath For Binary Access Read As #fileNumber
    fileLength = LOF(fileNumber)
    If fileLength = 0 Then
        Close #fileNumber
        Exit Function
    End If
    ReDim bytes(1 To fileLength)
    Get #fileNumber, , bytes
    Close #fileNumber

    startIndex = 1
    If fileLength >= 3 Then
        If bytes(1) = &HEF And bytes(2) = &HBB And bytes(3) = &HBF Then startIndex = 4
    End If

    index = startIndex
    Do While index <= fileLength
        b1 = CLng(bytes(index))
        If b1 < &H80 Then
            result = result & ChrWCode(b1)
            index = index + 1
        ElseIf (b1 And &HE0) = &HC0 And index + 1 <= fileLength Then
            b2 = CLng(bytes(index + 1))
            codePoint = ((b1 And &H1F) * &H40) + (b2 And &H3F)
            result = result & ChrWCode(codePoint)
            index = index + 2
        ElseIf (b1 And &HF0) = &HE0 And index + 2 <= fileLength Then
            b2 = CLng(bytes(index + 1))
            b3 = CLng(bytes(index + 2))
            codePoint = ((b1 And &HF) * &H1000) + ((b2 And &H3F) * &H40) + (b3 And &H3F)
            result = result & ChrWCode(codePoint)
            index = index + 3
        ElseIf (b1 And &HF8) = &HF0 And index + 3 <= fileLength Then
            b2 = CLng(bytes(index + 1))
            b3 = CLng(bytes(index + 2))
            b4 = CLng(bytes(index + 3))
            codePoint = ((b1 And &H7) * &H40000) + ((b2 And &H3F) * &H1000) + ((b3 And &H3F) * &H40) + (b4 And &H3F)
            result = result & ChrWCode(codePoint)
            index = index + 4
        Else
            Err.Raise 5, , "Invalid UTF-8 sequence in CSV file."
        End If
    Loop

    ReadUtf8TextFile = result
End Function

Private Function ChrWCode(ByVal codePoint As Long) As String
    Dim highSurrogate As Long
    Dim lowSurrogate As Long

    If codePoint < 0 Then Err.Raise 5, , "Invalid Unicode code point."
    If codePoint <= 65535 Then
        If codePoint > 32767 Then
            ChrWCode = ChrW$(codePoint - 65536)
        Else
            ChrWCode = ChrW$(codePoint)
        End If
    ElseIf codePoint <= 1114111 Then
        codePoint = codePoint - 65536
        highSurrogate = 55296 + (codePoint \ 1024)
        lowSurrogate = 56320 + (codePoint Mod 1024)
        ChrWCode = ChrWCode(highSurrogate) & ChrWCode(lowSurrogate)
    Else
        Err.Raise 5, , "Invalid Unicode code point."
    End If
End Function

Private Function ParseCsvText(ByVal csvText As String) As Collection
    Dim rows As Collection
    Dim fields As Collection
    Dim fieldText As String
    Dim index As Long
    Dim currentChar As String
    Dim nextChar As String
    Dim inQuotes As Boolean

    Set rows = New Collection
    Set fields = New Collection

    For index = 1 To Len(csvText)
        currentChar = Mid$(csvText, index, 1)
        If inQuotes Then
            If currentChar = """" Then
                If index < Len(csvText) Then
                    nextChar = Mid$(csvText, index + 1, 1)
                    If nextChar = """" Then
                        fieldText = fieldText & """"
                        index = index + 1
                    Else
                        inQuotes = False
                    End If
                Else
                    inQuotes = False
                End If
            Else
                fieldText = fieldText & currentChar
            End If
        Else
            If currentChar = """" Then
                inQuotes = True
            ElseIf currentChar = "," Then
                fields.Add fieldText
                fieldText = ""
            ElseIf currentChar = vbCr Or currentChar = vbLf Then
                fields.Add fieldText
                AddCsvRow rows, fields
                Set fields = New Collection
                fieldText = ""
                If currentChar = vbCr And index < Len(csvText) Then
                    If Mid$(csvText, index + 1, 1) = vbLf Then index = index + 1
                End If
            Else
                fieldText = fieldText & currentChar
            End If
        End If
    Next index

    If inQuotes Then Err.Raise 5, , "CSV contains an unclosed quoted field."
    If fieldText <> "" Or fields.Count > 0 Then
        fields.Add fieldText
        AddCsvRow rows, fields
    End If

    Set ParseCsvText = rows
End Function

Private Sub AddCsvRow(ByVal rows As Collection, ByVal fields As Collection)
    Dim fieldIndex As Long
    Dim values() As String
    Dim hasAnyValue As Boolean

    ReDim values(1 To fields.Count)
    For fieldIndex = 1 To fields.Count
        values(fieldIndex) = CStr(fields.Item(fieldIndex))
        If Trim$(values(fieldIndex)) <> "" Then hasAnyValue = True
    Next fieldIndex

    If hasAnyValue Or rows.Count = 0 Then rows.Add values
End Sub

Private Function FindCsvHeaderIndex(ByVal headerRow As Variant, ByVal requiredHeader As String) As Long
    Dim index As Long
    Dim normalizedRequired As String
    Dim normalizedHeader As String

    normalizedRequired = NormalizeCsvHeader(requiredHeader)
    For index = LBound(headerRow) To UBound(headerRow)
        normalizedHeader = NormalizeCsvHeader(CStr(headerRow(index)))
        If normalizedHeader = normalizedRequired Then
            FindCsvHeaderIndex = index
            Exit Function
        End If
    Next index
End Function

Private Function NormalizeCsvHeader(ByVal value As String) As String
    value = Replace(value, ChrWCode(65279), "")
    NormalizeCsvHeader = LCase$(Trim$(value))
End Function

Private Function CsvField(ByVal rowValues As Variant, ByVal fieldIndex As Long) As String
    If fieldIndex < LBound(rowValues) Or fieldIndex > UBound(rowValues) Then
        CsvField = ""
    Else
        CsvField = Trim$(CStr(rowValues(fieldIndex)))
    End If
End Function

Private Sub SyncTaskEstimateActionButtons(ByVal ws As Worksheet)

    Dim colIndex As Variant
    Dim actionCell As Range
    Dim shapeName As String
    Dim macroName As String

Private Function IsTaskEstimateActionCell(ByVal ws As Worksheet, ByVal Target As Range) As Boolean

        And Target.Column <> EST_SHEET_COL_ACTION_EXPRESS_EXPORT Then Exit Function

    actionCol = Target.Column
    IsTaskEstimateActionCell = (actionCol = EST_SHEET_COL_ACTION_DECOMPOSE _
        Or actionCol = EST_SHEET_COL_ACTION_DELETE _
        Or actionCol = EST_SHEET_COL_ACTION_EXPRESS_EXPORT)

SafeExit:
End Function

Public Function HandleTaskEstimateAction(ByVal ws As Worksheet, ByVal Target As Range) As Boolean
    On Error GoTo Fail

    Dim tbl As ListObject
    Dim actionCol As Long
    Dim rowIndex As Long

    Set tbl = ws.ListObjects("tblTaskEstimates")
    If tbl Is Nothing Then Err.Raise 9, , "Action cell is not inside a table."
    If tbl.Name <> "tblTaskEstimates" Then Err.Raise 9, , "Action cell is not inside tblTaskEstimates."
    If Target.Row < tbl.DataBodyRange.Row Or Target.Row > tbl.DataBodyRange.Row + tbl.DataBodyRange.Rows.Count - 1 Then Err.Raise 9, , "Action cell is outside task estimate row range."
    If Target.Column <> EST_SHEET_COL_ACTION_DECOMPOSE _
        And Target.Column <> EST_SHEET_COL_ACTION_DELETE _
        And Target.Column <> EST_SHEET_COL_ACTION_EXPRESS_EXPORT Then Err.Raise 5, , "Unsupported action column."
    If Trim$(CStr(Target.Value)) = "" Then Err.Raise 5, , "Task estimate action cell is empty."

    actionCol = Target.Column
    rowIndex = Target.Row - tbl.DataBodyRange.Row + 1

    Select Case actionCol
        Case EST_SHEET_COL_ACTION_DECOMPOSE
            DecomposeTaskEstimateRow tbl, rowIndex
        Case EST_SHEET_COL_ACTION_DELETE
            DeleteTaskEstimateDecomposition tbl, rowIndex, TASK_DELETE_CHOICE_PROMPT
        Case EST_SHEET_COL_ACTION_EXPRESS_EXPORT
            ExportTaskEstimateExpressWithPrompt tbl, rowIndex
        Case Else
            Err.Raise 5, , "Unsupported task estimate action."
    End Select

    HandleTaskEstimateAction = True
    Exit Function

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "HandleTaskEstimateAction: " & CStr(Err.Number) & " - " & Err.Description & vbCrLf & _
        "Sheet: " & ws.Name & vbCrLf & _
        "Cell: " & Target.Address(False, False), vbExclamation
End Function

Private Sub RefreshTaskEstimateTable(ByVal tbl As ListObject)
    On Error GoTo Fail

    Dim rowIndex As Long
    Dim taskLevel As Long
    Dim actionRange As Range
    Dim hasChildren As Boolean
    Dim childParentIds() As String
    Dim childParentCount As Long
    Dim taskId As String
    Dim parentTaskId As String
    Dim currentMaxRow As Long
    Dim refreshMaxRow As Long
    Dim refreshStartRow As Long

    If tbl.DataBodyRange Is Nothing Then Exit Sub

    currentMaxRow = LastTaskEstimateUsedRow(tbl)
    refreshMaxRow = currentMaxRow + 1
    If mTaskEstimateMaxActionRow + 1 > refreshMaxRow Then refreshMaxRow = mTaskEstimateMaxActionRow + 1
    If refreshMaxRow < 1 Then refreshMaxRow = 1
    If refreshMaxRow > tbl.DataBodyRange.Rows.Count Then refreshMaxRow = tbl.DataBodyRange.Rows.Count
    refreshStartRow = mTaskEstimateRefreshStartRow
    If refreshStartRow < 1 Then refreshStartRow = 1
    If refreshStartRow > refreshMaxRow Then refreshStartRow = refreshMaxRow

    ReDim childParentIds(1 To tbl.DataBodyRange.Rows.Count)
    For rowIndex = 1 To currentMaxRow
        If TaskEstimateRowHasTask(tbl, rowIndex) Then
            parentTaskId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value))
            If parentTaskId <> "" Then
                childParentCount = childParentCount + 1
                childParentIds(childParentCount) = parentTaskId
            End If
        End If
    Next rowIndex

    Set actionRange = tbl.Parent.Range(tbl.Parent.Cells(tbl.DataBodyRange.Row + refreshStartRow - 1, EST_SHEET_COL_ACTION_DECOMPOSE), tbl.Parent.Cells(tbl.DataBodyRange.Row + refreshMaxRow - 1, EST_SHEET_COL_ACTION_DELETE))
    actionRange.Hyperlinks.Delete
    tbl.Parent.Range(tbl.Parent.Cells(tbl.DataBodyRange.Row + refreshStartRow - 1, EST_SHEET_COL_ACTION_EXPRESS_EXPORT), tbl.Parent.Cells(tbl.DataBodyRange.Row + refreshMaxRow - 1, EST_SHEET_COL_ACTION_EXPRESS_EXPORT)).Hyperlinks.Delete

    For rowIndex = refreshStartRow To refreshMaxRow
        TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DECOMPOSE).ClearContents
        TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DELETE).ClearContents
        TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_EXPRESS_EXPORT).ClearContents

        If TaskEstimateRowHasTask(tbl, rowIndex) Then
            taskId = EnsureTaskEstimateRowId(tbl, rowIndex)
            taskLevel = TaskEstimateRowLevel(tbl, rowIndex)
            TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_LEVEL).Value = taskLevel
            tbl.DataBodyRange.Cells(rowIndex, EST_COL_DESCRIPTION).IndentLevel = taskLevel
            hasChildren = TaskEstimateIdInList(taskId, childParentIds, childParentCount)

            If taskLevel < EST_MAX_DECOMPOSITION_LEVEL Then
                TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DECOMPOSE).Value = TaskEstimateTextDecompose()
            End If
            If hasChildren Or taskLevel > 0 Then
                TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DELETE).Value = TaskEstimateTextDeleteDecomposition()
            End If
            If taskLevel = 0 Then
                TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_EXPRESS_EXPORT).Value = TaskEstimateTextExpressExport()
            End If
            If hasChildren Then
                WriteTaskEstimateParentFormulas tbl, rowIndex
                ApplyTaskEstimateCalculatedStyle tbl, rowIndex
            Else
                ApplyTaskEstimateInputStyle tbl, rowIndex
            End If
        Else
            ClearTaskEstimateRow tbl, rowIndex
        End If

        AddTaskEstimateActionLink TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DECOMPOSE)
        AddTaskEstimateActionLink TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DELETE)
        AddTaskEstimateActionLink TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_EXPRESS_EXPORT)
    Next rowIndex
    Exit Sub

Fail:
    MsgBox TaskEstimateTextErrorPrefix() & "RefreshTaskEstimateTable: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
End Sub

Private Sub WriteTaskEstimateParentFormula(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal estimateCol As Long)
    Dim targetCell As Range
    Dim parentIdCell As Range
    Dim parentIdRange As Range
    Dim sumRange As Range
    Dim sumExpression As String

    Set targetCell = tbl.DataBodyRange.Cells(rowIndex, estimateCol)
    Set parentIdCell = TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID)
    Set parentIdRange = tbl.Parent.Range( _
        tbl.Parent.Cells(tbl.DataBodyRange.Row, EST_SHEET_COL_PARENT_TASK_ID), _
        tbl.Parent.Cells(tbl.DataBodyRange.Row + tbl.DataBodyRange.Rows.Count - 1, EST_SHEET_COL_PARENT_TASK_ID))
    Set sumRange = tbl.Parent.Range( _
        tbl.DataBodyRange.Cells(1, estimateCol), _

Private Sub DecomposeTaskEstimateRow(ByVal tbl As ListObject, ByVal rowIndex As Long)
    Dim taskLevel As Long
    Dim insertRowIndex As Long
    Dim col As Long
    Dim parentTaskId As String
    Dim hadChildren As Boolean

    If rowIndex < 1 Or rowIndex > tbl.DataBodyRange.Rows.Count Then Err.Raise 9, , "Task row index is outside table."
    If Not TaskEstimateRowHasTask(tbl, rowIndex) Then Err.Raise 5, , "Cannot decompose an empty row."

    taskLevel = TaskEstimateRowLevel(tbl, rowIndex)
    If taskLevel >= EST_MAX_DECOMPOSITION_LEVEL Then
        Err.Raise 5, , "Cannot decompose below level " & CStr(EST_MAX_DECOMPOSITION_LEVEL) & "."
    End If

    If TaskEstimateRowHasTask(tbl, tbl.DataBodyRange.Rows.Count) Then
        Err.Raise 5, , "Task estimate table has no free rows for decomposition."
    End If

    parentTaskId = EnsureTaskEstimateRowId(tbl, rowIndex)
    hadChildren = TaskEstimateRowHasChildren(tbl, rowIndex)
    insertRowIndex = LastTaskEstimateDescendantRow(tbl, rowIndex) + 1
    If insertRowIndex > tbl.DataBodyRange.Rows.Count Then
        Err.Raise 5, , "Task estimate table has no free row below the selected task."
    End If

    ShiftTaskEstimateRowsDown tbl, insertRowIndex

    For col = EST_COL_PRIORITY To EST_COL_NOTE
        tbl.DataBodyRange.Cells(insertRowIndex, col).Value = tbl.DataBodyRange.Cells(rowIndex, col).Value
    Next col
    For col = EST_COL_AN To EST_COL_QA
        If hadChildren Then
            tbl.DataBodyRange.Cells(insertRowIndex, col).ClearContents
        Else
            tbl.DataBodyRange.Cells(insertRowIndex, col).Value = tbl.DataBodyRange.Cells(rowIndex, col).Value
            tbl.DataBodyRange.Cells(rowIndex, col).ClearContents
        End If
    Next col
    tbl.DataBodyRange.Cells(insertRowIndex, EST_COL_COMMENT).Value = tbl.DataBodyRange.Cells(rowIndex, EST_COL_COMMENT).Value
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_TASK_ID).Value = NewTaskEstimateId()
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value = parentTaskId
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_LEVEL).Value = taskLevel + 1

    WriteTaskEstimateParentFormulas tbl, rowIndex
    ApplyTaskEstimateCalculatedStyle tbl, rowIndex
    ApplyTaskEstimateInputStyle tbl, insertRowIndex
    MarkTaskEstimateRefreshFrom rowIndex
End Sub

Private Sub DeleteTaskEstimateDecomposition(ByVal tbl As ListObject, ByVal rowIndex As Long, Optional ByVal deleteChoice As Long = TASK_DELETE_CHOICE_PROMPT)
    Dim lastDescendantRow As Long
    Dim deleteCount As Long
    Dim deleteLastRow As Long
    Dim anSum As Double
    Dim beSum As Double
    Dim feSum As Double
    Dim qaSum As Double
    Dim transferRowIndex As Long
    Dim writableRowIndex As Long
    Dim writeToParentAfterDelete As Boolean

    If rowIndex < 1 Or rowIndex > tbl.DataBodyRange.Rows.Count Then Err.Raise 9, , "Task row index is outside table."
    If Not TaskEstimateRowHasTask(tbl, rowIndex) Then Err.Raise 5, , "Cannot delete decomposition for an empty row."
    If TaskEstimateRowLevel(tbl, rowIndex) = 0 And Not TaskEstimateRowHasChildren(tbl, rowIndex) Then Err.Raise 5, , "Root task without decomposition cannot be deleted with the decomposition delete action."
    MarkTaskEstimateRefreshFrom TaskEstimateAncestorRefreshRow(tbl, rowIndex)

    lastDescendantRow = LastTaskEstimateDescendantRow(tbl, rowIndex)
    deleteLastRow = lastDescendantRow
    deleteCount = deleteLastRow - rowIndex + 1

    If deleteChoice = TASK_DELETE_CHOICE_PROMPT Then
        deleteChoice = PromptTaskEstimateDeleteChoice(tbl, rowIndex)
    End If
    If deleteChoice = TASK_DELETE_CHOICE_PROMPT Then Exit Sub
    If deleteChoice < TASK_DELETE_CHOICE_DROP Or deleteChoice > TASK_DELETE_CHOICE_DOWN Then Err.Raise 5, , "Unsupported delete choice: " & CStr(deleteChoice)

    SumTaskEstimateDeleteEstimates tbl, rowIndex, anSum, beSum, feSum, qaSum

    If deleteChoice <> TASK_DELETE_CHOICE_DROP Then
        transferRowIndex = FindTaskEstimateSiblingRow(tbl, rowIndex, deleteChoice)
        If transferRowIndex = 0 Then
            If TaskEstimateRowLevel(tbl, rowIndex) = 0 Then
                Err.Raise 5, , "Root task has no sibling in selected transfer direction."
            End If
            transferRowIndex = FindTaskEstimateRowById(tbl, Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value)))
            If transferRowIndex = 0 Then Err.Raise 9, , "Parent task not found for fallback transfer."
        End If
        If transferRowIndex = 0 Then Err.Raise 5, , "No valid transfer target for selected delete action."

        writableRowIndex = ResolveTaskEstimateWritableTransferRow(tbl, transferRowIndex, deleteChoice, rowIndex, deleteLastRow)
        If writableRowIndex > 0 Then
            AddTaskEstimateSumsToRow tbl, writableRowIndex, anSum, beSum, feSum, qaSum
        Else
            writeToParentAfterDelete = True
        End If
    End If

    DeleteTaskEstimateRows tbl, rowIndex, deleteCount

    If writeToParentAfterDelete Then
        AddTaskEstimateSumsToRow tbl, AdjustTaskEstimateRowAfterDelete(transferRowIndex, rowIndex, deleteCount), anSum, beSum, feSum, qaSum
    End If

End Sub

Private Function PromptTaskEstimateDeleteChoice(ByVal tbl As ListObject, ByVal rowIndex As Long) As Long
    Dim rawValue As Variant
    Dim promptText As String

    promptText = "Удалить декомпозированную задачу?" & vbCrLf & vbCrLf & _
        "1 - удалить вместе с оценками" & vbCrLf & _
        "2 - перенести оценки вверх и удалить" & vbCrLf & _
        "3 - перенести оценки вниз и удалить" & vbCrLf & vbCrLf & _
        "Строка: " & CStr(tbl.DataBodyRange.Row + rowIndex - 1)

    Do
        rawValue = Application.InputBox(promptText, "Удаление декомпозиции", Type:=1)
        If VarType(rawValue) = vbBoolean Then
            If rawValue = False Then
                PromptTaskEstimateDeleteChoice = TASK_DELETE_CHOICE_PROMPT
                Exit Function
            End If
        End If
        If IsNumeric(rawValue) Then
            If CLng(rawValue) >= TASK_DELETE_CHOICE_DROP And CLng(rawValue) <= TASK_DELETE_CHOICE_DOWN Then
                PromptTaskEstimateDeleteChoice = CLng(rawValue)
                Exit Function
            End If
        End If
        MsgBox "Введите 1, 2 или 3.", vbExclamation
    Loop
End Function

Private Sub SumTaskEstimateDeleteEstimates(ByVal tbl As ListObject, ByVal rowIndex As Long, ByRef anSum As Double, ByRef beSum As Double, ByRef feSum As Double, ByRef qaSum As Double)
    If TaskEstimateRowHasChildren(tbl, rowIndex) Then
        SumTaskEstimateLeafEstimates tbl, rowIndex, anSum, beSum, feSum, qaSum
    Else
        anSum = NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_AN).Value)
        beSum = NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_BE).Value)
        feSum = NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_FE).Value)
        qaSum = NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_QA).Value)
    End If
End Sub

Private Function ResolveTaskEstimateTransferRow(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal deleteChoice As Long) As Long
    Dim siblingRow As Long
    Dim parentId As String

    siblingRow = FindTaskEstimateSiblingRow(tbl, rowIndex, deleteChoice)
    If siblingRow > 0 Then
        ResolveTaskEstimateTransferRow = siblingRow
        Exit Function
    End If

    If TaskEstimateRowLevel(tbl, rowIndex) = 0 Then
        Err.Raise 5, , "Root task has no sibling in selected transfer direction."
    End If

    parentId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value))
    ResolveTaskEstimateTransferRow = FindTaskEstimateRowById(tbl, parentId)
    If ResolveTaskEstimateTransferRow = 0 Then Err.Raise 9, , "Parent task not found for fallback transfer."
End Function

Private Function FindTaskEstimateSiblingRow(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal deleteChoice As Long) As Long
    Dim sourceLevel As Long
    Dim sourceParentId As String
    Dim scanRow As Long
    Dim scanLevel As Long

    sourceLevel = TaskEstimateRowLevel(tbl, rowIndex)
    sourceParentId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value))

    If deleteChoice = TASK_DELETE_CHOICE_UP Then
        For scanRow = rowIndex - 1 To 1 Step -1
            If TaskEstimateRowHasTask(tbl, scanRow) Then
                scanLevel = TaskEstimateRowLevel(tbl, scanRow)
                If scanLevel < sourceLevel Then Exit For
                If scanLevel = sourceLevel And Trim$(CStr(TaskEstimateSheetCell(tbl, scanRow, EST_SHEET_COL_PARENT_TASK_ID).Value)) = sourceParentId Then
                    FindTaskEstimateSiblingRow = scanRow
                    Exit Function
                End If
            End If
        Next scanRow
    ElseIf deleteChoice = TASK_DELETE_CHOICE_DOWN Then
        scanRow = LastTaskEstimateDescendantRow(tbl, rowIndex) + 1
        Do While scanRow <= tbl.DataBodyRange.Rows.Count
            If Not TaskEstimateRowHasTask(tbl, scanRow) Then Exit Do
            scanLevel = TaskEstimateRowLevel(tbl, scanRow)
            If scanLevel < sourceLevel Then Exit Do
            If scanLevel = sourceLevel Then
                If Trim$(CStr(TaskEstimateSheetCell(tbl, scanRow, EST_SHEET_COL_PARENT_TASK_ID).Value)) = sourceParentId Then
                    FindTaskEstimateSiblingRow = scanRow
                End If
                Exit Function
            End If
            scanRow = LastTaskEstimateDescendantRow(tbl, scanRow) + 1
        Loop
    End If
End Function

Private Function ResolveTaskEstimateWritableTransferRow(ByVal tbl As ListObject, ByVal transferRowIndex As Long, ByVal deleteChoice As Long, ByVal deleteFirstRow As Long, ByVal deleteLastRow As Long) As Long
    Dim candidateRow As Long

    If transferRowIndex <= 0 Then Exit Function
    If Not TaskEstimateRowHasChildren(tbl, transferRowIndex) Then
        ResolveTaskEstimateWritableTransferRow = transferRowIndex
        Exit Function
    End If

    candidateRow = FindTaskEstimateLeafDescendant(tbl, transferRowIndex, deleteChoice, deleteFirstRow, deleteLastRow)
    If candidateRow > 0 Then
        ResolveTaskEstimateWritableTransferRow = candidateRow
    End If
End Function

Private Function FindTaskEstimateLeafDescendant(ByVal tbl As ListObject, ByVal parentRowIndex As Long, ByVal deleteChoice As Long, ByVal deleteFirstRow As Long, ByVal deleteLastRow As Long) As Long
    Dim firstRow As Long
    Dim lastRow As Long
    Dim scanRow As Long

    firstRow = parentRowIndex + 1
    lastRow = LastTaskEstimateDescendantRow(tbl, parentRowIndex)
    If lastRow < firstRow Then Exit Function

    If deleteChoice = TASK_DELETE_CHOICE_UP Then
        For scanRow = lastRow To firstRow Step -1
            If scanRow < deleteFirstRow Or scanRow > deleteLastRow Then
                If TaskEstimateRowHasTask(tbl, scanRow) And Not TaskEstimateRowHasChildren(tbl, scanRow) Then
                    FindTaskEstimateLeafDescendant = scanRow
                    Exit Function
                End If
            End If
        Next scanRow
    Else
        For scanRow = firstRow To lastRow
            If scanRow < deleteFirstRow Or scanRow > deleteLastRow Then
                If TaskEstimateRowHasTask(tbl, scanRow) And Not TaskEstimateRowHasChildren(tbl, scanRow) Then
                    FindTaskEstimateLeafDescendant = scanRow
                    Exit Function
                End If
            End If
        Next scanRow
    End If
End Function

Private Sub AddTaskEstimateSumsToRow(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal anSum As Double, ByVal beSum As Double, ByVal feSum As Double, ByVal qaSum As Double)
    If rowIndex < 1 Or rowIndex > tbl.DataBodyRange.Rows.Count Then Err.Raise 9, , "Transfer target row is outside task estimate table."
    If TaskEstimateRowHasChildren(tbl, rowIndex) Then Err.Raise 5, , "Cannot write transferred estimates directly into a parent formula row."

    WriteEstimateNumber tbl.DataBodyRange.Cells(rowIndex, EST_COL_AN), NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_AN).Value) + anSum
    WriteEstimateNumber tbl.DataBodyRange.Cells(rowIndex, EST_COL_BE), NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_BE).Value) + beSum
    WriteEstimateNumber tbl.DataBodyRange.Cells(rowIndex, EST_COL_FE), NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_FE).Value) + feSum
    WriteEstimateNumber tbl.DataBodyRange.Cells(rowIndex, EST_COL_QA), NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_QA).Value) + qaSum
End Sub

Private Function AdjustTaskEstimateRowAfterDelete(ByVal rowIndex As Long, ByVal deleteFirstRow As Long, ByVal deleteCount As Long) As Long
    If rowIndex > deleteFirstRow Then
        AdjustTaskEstimateRowAfterDelete = rowIndex - deleteCount
    Else
        AdjustTaskEstimateRowAfterDelete = rowIndex
    End If
End Function

Private Sub ShiftTaskEstimateRowsDown(ByVal tbl As ListObject, ByVal insertRowIndex As Long)
    Dim col As Long
    Dim lastRowIndex As Long
    Dim moveCount As Long
    Dim visibleValues As Variant
    Dim hiddenValues As Variant
    Dim ws As Worksheet

    Set ws = tbl.Parent
    lastRowIndex = tbl.DataBodyRange.Rows.Count
    moveCount = lastRowIndex - insertRowIndex

    If moveCount > 0 Then
        visibleValues = tbl.Parent.Range( _
            tbl.DataBodyRange.Cells(insertRowIndex, EST_COL_PRIORITY), _
            tbl.DataBodyRange.Cells(lastRowIndex - 1, EST_COL_LAST)).Value
        tbl.Parent.Range( _
            tbl.DataBodyRange.Cells(insertRowIndex + 1, EST_COL_PRIORITY), _
            tbl.DataBodyRange.Cells(lastRowIndex, EST_COL_LAST)).Value = visibleValues

        hiddenValues = ws.Range( _
            ws.Cells(tbl.DataBodyRange.Row + insertRowIndex - 1, EST_SHEET_COL_TASK_ID), _
            ws.Cells(tbl.DataBodyRange.Row + lastRowIndex - 2, EST_SHEET_COL_LEVEL)).Value
        ws.Range( _
            ws.Cells(tbl.DataBodyRange.Row + insertRowIndex, EST_SHEET_COL_TASK_ID), _
            ws.Cells(tbl.DataBodyRange.Row + lastRowIndex - 1, EST_SHEET_COL_LEVEL)).Value = hiddenValues
    End If

    For col = EST_COL_PRIORITY To EST_COL_LAST
        tbl.DataBodyRange.Cells(insertRowIndex, col).ClearContents
    Next col
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_TASK_ID).ClearContents
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_PARENT_TASK_ID).ClearContents
    TaskEstimateSheetCell(tbl, insertRowIndex, EST_SHEET_COL_LEVEL).ClearContents
End Sub

Private Sub DeleteTaskEstimateRows(ByVal tbl As ListObject, ByVal firstRowIndex As Long, ByVal deleteCount As Long)
    Dim rowIndex As Long
    Dim lastRowIndex As Long
    Dim moveCount As Long
    Dim visibleValues As Variant
    Dim hiddenValues As Variant
    Dim ws As Worksheet

    Set ws = tbl.Parent
    lastRowIndex = tbl.DataBodyRange.Rows.Count
    moveCount = lastRowIndex - firstRowIndex - deleteCount + 1

    If moveCount > 0 Then
        visibleValues = tbl.Parent.Range( _
            tbl.DataBodyRange.Cells(firstRowIndex + deleteCount, EST_COL_PRIORITY), _
            tbl.DataBodyRange.Cells(lastRowIndex, EST_COL_LAST)).Value
        tbl.Parent.Range( _
            tbl.DataBodyRange.Cells(firstRowIndex, EST_COL_PRIORITY), _
            tbl.DataBodyRange.Cells(lastRowIndex - deleteCount, EST_COL_LAST)).Value = visibleValues

        hiddenValues = ws.Range( _
            ws.Cells(tbl.DataBodyRange.Row + firstRowIndex + deleteCount - 1, EST_SHEET_COL_TASK_ID), _
            ws.Cells(tbl.DataBodyRange.Row + lastRowIndex - 1, EST_SHEET_COL_LEVEL)).Value
        ws.Range( _
            ws.Cells(tbl.DataBodyRange.Row + firstRowIndex - 1, EST_SHEET_COL_TASK_ID), _
            ws.Cells(tbl.DataBodyRange.Row + lastRowIndex - deleteCount - 1, EST_SHEET_COL_LEVEL)).Value = hiddenValues
    End If

    For rowIndex = tbl.DataBodyRange.Rows.Count - deleteCount + 1 To tbl.DataBodyRange.Rows.Count
        ClearTaskEstimateRow tbl, rowIndex
    Next rowIndex
End Sub

Private Sub ClearTaskEstimateRow(ByVal tbl As ListObject, ByVal rowIndex As Long)
    Dim col As Long

    TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DECOMPOSE).ClearContents
    TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_ACTION_DELETE).ClearContents
    For col = EST_COL_PRIORITY To EST_COL_LAST
        tbl.DataBodyRange.Cells(rowIndex, col).ClearContents
    Next col
    TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID).ClearContents
    TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).ClearContents
    TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_LEVEL).ClearContents
    tbl.DataBodyRange.Cells(rowIndex, EST_COL_DESCRIPTION).IndentLevel = 0
End Sub

Private Function TaskEstimateRowHasTask(ByVal tbl As ListObject, ByVal rowIndex As Long) As Boolean
    Dim col As Variant
    Dim valueText As String

    If tbl.DataBodyRange Is Nothing Then Exit Function
    If rowIndex < 1 Or rowIndex > tbl.DataBodyRange.Rows.Count Then Exit Function

    For Each col In Array(EST_COL_PRIORITY, EST_COL_DIRECTION, EST_COL_DESCRIPTION, EST_COL_TICKET, EST_COL_NOTE, EST_COL_AN, EST_COL_BE, EST_COL_FE, EST_COL_QA)
        valueText = Trim$(CStr(tbl.DataBodyRange.Cells(rowIndex, CLng(col)).Value))
        If valueText <> "" Then
            TaskEstimateRowHasTask = True
            Exit Function
        End If
    Next col
End Function

Private Function TaskEstimateRowLevel(ByVal tbl As ListObject, ByVal rowIndex As Long) As Long
    Dim rawValue As Variant
    Dim result As Long

    rawValue = TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_LEVEL).Value
    If IsNumeric(rawValue) Then result = CLng(rawValue)
    If result < 0 Then result = 0
    If result > EST_MAX_DECOMPOSITION_LEVEL Then result = EST_MAX_DECOMPOSITION_LEVEL
    TaskEstimateRowLevel = result
End Function

Private Function EnsureTaskEstimateRowId(ByVal tbl As ListObject, ByVal rowIndex As Long) As String
    Dim taskId As String

    taskId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID).Value))
    If taskId = "" Then
        taskId = NewTaskEstimateId()
        TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID).Value = taskId
    End If
    EnsureTaskEstimateRowId = taskId
End Function

Private Function NewTaskEstimateId() As String
    Randomize
    NewTaskEstimateId = "t" & Format$(Now, "yyyymmddhhmmss") & "_" & CStr(Int(Rnd() * 1000000#))
End Function

Private Function FindTaskEstimateRowById(ByVal tbl As ListObject, ByVal taskId As String) As Long
    Dim rowIndex As Long
    Dim normalizedTaskId As String

    normalizedTaskId = Trim$(taskId)
    If normalizedTaskId = "" Then Exit Function

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        If Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID).Value)) = normalizedTaskId Then
            FindTaskEstimateRowById = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function TaskEstimateAncestorRefreshRow(ByVal tbl As ListObject, ByVal rowIndex As Long) As Long
    Dim result As Long
    Dim parentId As String
    Dim parentRow As Long

    result = rowIndex
    parentId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_PARENT_TASK_ID).Value))
    Do While parentId <> ""
        parentRow = FindTaskEstimateRowById(tbl, parentId)
        If parentRow = 0 Then Exit Do
        If parentRow < result Then result = parentRow
        parentId = Trim$(CStr(TaskEstimateSheetCell(tbl, parentRow, EST_SHEET_COL_PARENT_TASK_ID).Value))
    Loop
    TaskEstimateAncestorRefreshRow = result
End Function

Private Function TaskEstimateRowHasChildren(ByVal tbl As ListObject, ByVal rowIndex As Long) As Boolean
    Dim taskId As String
    Dim scanRow As Long

    taskId = Trim$(CStr(TaskEstimateSheetCell(tbl, rowIndex, EST_SHEET_COL_TASK_ID).Value))
    If taskId = "" Then Exit Function

    For scanRow = 1 To tbl.DataBodyRange.Rows.Count
        If scanRow <> rowIndex Then
            If Trim$(CStr(TaskEstimateSheetCell(tbl, scanRow, EST_SHEET_COL_PARENT_TASK_ID).Value)) = taskId _
                And TaskEstimateRowHasTask(tbl, scanRow) Then
                TaskEstimateRowHasChildren = True
                Exit Function
            End If
        End If
    Next scanRow
End Function

Private Function LastTaskEstimateDescendantRow(ByVal tbl As ListObject, ByVal rowIndex As Long) As Long
    Dim baseLevel As Long
    Dim scanRow As Long
    Dim scanLevel As Long

    baseLevel = TaskEstimateRowLevel(tbl, rowIndex)
    LastTaskEstimateDescendantRow = rowIndex

    For scanRow = rowIndex + 1 To tbl.DataBodyRange.Rows.Count
        If Not TaskEstimateRowHasTask(tbl, scanRow) Then Exit For
        scanLevel = TaskEstimateRowLevel(tbl, scanRow)
        If scanLevel <= baseLevel Then Exit For
        LastTaskEstimateDescendantRow = scanRow
    Next scanRow
End Function

Private Sub SumTaskEstimateLeafEstimates(ByVal tbl As ListObject, ByVal rootRowIndex As Long, ByRef anSum As Double, ByRef beSum As Double, ByRef feSum As Double, ByRef qaSum As Double)
    Dim lastDescendantRow As Long
    Dim rowIndex As Long

    lastDescendantRow = LastTaskEstimateDescendantRow(tbl, rootRowIndex)
    For rowIndex = rootRowIndex + 1 To lastDescendantRow
        If TaskEstimateRowHasTask(tbl, rowIndex) And Not TaskEstimateRowHasChildren(tbl, rowIndex) Then
            anSum = anSum + NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_AN).Value)
            beSum = beSum + NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_BE).Value)
            feSum = feSum + NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_FE).Value)
            qaSum = qaSum + NzDouble(tbl.DataBodyRange.Cells(rowIndex, EST_COL_QA).Value)
        End If
    Next rowIndex
End Sub

Private Function EffectiveTaskEstimateValue(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal estimateCol As Long) As Double
    Dim anSum As Double
    Dim beSum As Double
    Dim feSum As Double
    Dim qaSum As Double

    If TaskEstimateRowHasChildren(tbl, rowIndex) Then
        SumTaskEstimateLeafEstimates tbl, rowIndex, anSum, beSum, feSum, qaSum
        Select Case estimateCol
            Case EST_COL_AN
                EffectiveTaskEstimateValue = anSum
            Case EST_COL_BE
                EffectiveTaskEstimateValue = beSum
            Case EST_COL_FE
                EffectiveTaskEstimateValue = feSum
            Case EST_COL_QA
                EffectiveTaskEstimateValue = qaSum
        End Select
    Else
        EffectiveTaskEstimateValue = NzDouble(tbl.DataBodyRange.Cells(rowIndex, estimateCol).Value)
    End If
End Function

Private Sub WriteEstimateNumber(ByVal targetCell As Range, ByVal value As Double)
    If value = 0 Then
        targetCell.ClearContents
    Else
        targetCell.Value = value
    End If
End Sub

Private Function TaskEstimateSheetCell(ByVal tbl As ListObject, ByVal rowIndex As Long, ByVal sheetCol As Long) As Range
    Set TaskEstimateSheetCell = tbl.Parent.Cells(tbl.DataBodyRange.Row + rowIndex - 1, sheetCol)
End Function

Private Function IsQuarterPlanActionCell(ByVal ws As Worksheet, ByVal Target As Range) As Boolean
    On Error GoTo SafeExit

    Dim tbl As ListObject
    Dim actionCol As Long

    Set tbl = GetContainingTable(ws, Target)
    If tbl Is Nothing Then Exit Function
    If Intersect(Target, tbl.DataBodyRange) Is Nothing Then Exit Function
    If Target.Column < tbl.Range.Column Or Target.Column > tbl.Range.Column + PLAN_COL_ACTION_DOWN - 1 Then Exit Function
    If Trim$(CStr(Target.Value)) = "" Then Exit Function

    actionCol = Target.Column - tbl.Range.Column + 1

    Select Case tbl.Name
        Case "tblPlanBacklog"
            IsQuarterPlanActionCell = (actionCol = PLAN_COL_ACTION_TO_PLAN _
                Or actionCol = PLAN_COL_ACTION_UP _
                Or actionCol = PLAN_COL_ACTION_DOWN)
        Case "tblPlanActive"
            IsQuarterPlanActionCell = (actionCol = PLAN_COL_ACTION_TO_GREY _
                Or actionCol = PLAN_COL_ACTION_TO_BACKLOG _
                Or actionCol = PLAN_COL_ACTION_UP _
                Or actionCol = PLAN_COL_ACTION_DOWN)
        Case "tblPlanGrey"
            IsQuarterPlanActionCell = (actionCol = PLAN_COL_ACTION_TO_BACKLOG _
                Or actionCol = PLAN_COL_ACTION_UP _
                Or actionCol = PLAN_COL_ACTION_DOWN)
    End Select

SafeExit:
End Function

Public Function HandleQuarterPlanAction(ByVal ws As Worksheet, ByVal Target As Range) As Boolean
    On Error GoTo SafeExit

    Dim tbl As ListObject
    Dim actionCol As Long
    Dim sourceName As String
    Dim targetName As String

    Set tbl = GetContainingTable(ws, Target)
    If tbl Is Nothing Then Exit Function
    If Intersect(Target, tbl.DataBodyRange) Is Nothing Then Exit Function
    If Target.Column < tbl.Range.Column Or Target.Column > tbl.Range.Column + PLAN_COL_ACTION_DOWN - 1 Then Exit Function
    If Trim$(CStr(Target.Value)) = "" Then Exit Function

    sourceName = tbl.Name
    actionCol = Target.Column - tbl.Range.Column + 1

    If actionCol = PLAN_COL_ACTION_UP Or actionCol = PLAN_COL_ACTION_DOWN Then
        Application.EnableEvents = False
        HandleQuarterPlanAction = ReorderQuarterPlanTask(tbl, Target.Row - tbl.DataBodyRange.Row + 1, actionCol)
        Application.EnableEvents = True
        Exit Function
    End If

    Select Case sourceName
        Case "tblPlanBacklog"
            If actionCol <> PLAN_COL_ACTION_TO_PLAN Then Exit Function
            targetName = "tblPlanActive"
        Case "tblPlanActive"
            If actionCol = PLAN_COL_ACTION_TO_GREY Then
                targetName = "tblPlanGrey"
            ElseIf actionCol = PLAN_COL_ACTION_TO_BACKLOG Then
                targetName = "tblPlanBacklog"
            Else
                Exit Function
            End If
        Case "tblPlanGrey"
            If actionCol <> PLAN_COL_ACTION_TO_BACKLOG Then Exit Function
            targetName = "tblPlanBacklog"
        Case Else
            Exit Function
    End Select

    Application.EnableEvents = False
    MoveQuarterPlanTask ws, tbl, ws.ListObjects(targetName), Target.Row - tbl.DataBodyRange.Row + 1
    Application.EnableEvents = True
    HandleQuarterPlanAction = True

SafeExit:
    Application.EnableEvents = True
End Function

Private Function ReorderQuarterPlanTask(ByVal tbl As ListObject, ByVal sourceRowIndex As Long, ByVal actionCol As Long) As Boolean
    Dim targetRowIndex As Long

    If sourceRowIndex < 1 Or sourceRowIndex > tbl.DataBodyRange.Rows.Count Then Exit Function
    If Not QuarterPlanRowHasTask(tbl, sourceRowIndex) Then Exit Function

    If actionCol = PLAN_COL_ACTION_UP Then
        targetRowIndex = PreviousTaskQuarterPlanRow(tbl, sourceRowIndex)
    ElseIf actionCol = PLAN_COL_ACTION_DOWN Then
        targetRowIndex = NextTaskQuarterPlanRow(tbl, sourceRowIndex)
    End If

    If targetRowIndex = 0 Then Exit Function

    SwapQuarterPlanRows tbl, sourceRowIndex, targetRowIndex
    RefreshQuarterPlanTable tbl
    SyncQuarterPlanTableActionButtons tbl
    ReorderQuarterPlanTask = True
End Function

Private Sub SwapQuarterPlanRows(ByVal tbl As ListObject, ByVal firstRowIndex As Long, ByVal secondRowIndex As Long)
    Dim firstValues As Variant
    Dim secondValues As Variant
    Dim firstTechValues As Variant
    Dim secondTechValues As Variant
    Dim firstSheetRow As Long
    Dim secondSheetRow As Long

    firstValues = tbl.DataBodyRange.Cells(firstRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value
    secondValues = tbl.DataBodyRange.Cells(secondRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value
    tbl.DataBodyRange.Cells(firstRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value = secondValues
    tbl.DataBodyRange.Cells(secondRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value = firstValues

    firstSheetRow = tbl.DataBodyRange.Row + firstRowIndex - 1
    secondSheetRow = tbl.DataBodyRange.Row + secondRowIndex - 1

    firstTechValues = tbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & firstSheetRow & ":" & PLAN_TECH_LAST_COL_LETTER & firstSheetRow).Value
    secondTechValues = tbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & secondSheetRow & ":" & PLAN_TECH_LAST_COL_LETTER & secondSheetRow).Value
    tbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & firstSheetRow & ":" & PLAN_TECH_LAST_COL_LETTER & firstSheetRow).Value = secondTechValues
    tbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & secondSheetRow & ":" & PLAN_TECH_LAST_COL_LETTER & secondSheetRow).Value = firstTechValues
End Sub

Private Function GetContainingTable(ByVal ws As Worksheet, ByVal Target As Range) As ListObject
    Dim tbl As ListObject

    For Each tbl In ws.ListObjects
        If Not tbl.DataBodyRange Is Nothing Then
            If Not Intersect(Target, tbl.DataBodyRange) Is Nothing Then
                Set GetContainingTable = tbl
                Exit Function
            End If
        End If
    Next tbl
End Function

Private Sub MoveQuarterPlanTask(ByVal ws As Worksheet, ByVal sourceTbl As ListObject, ByVal targetTbl As ListObject, ByVal sourceRowIndex As Long)
    Dim targetRowIndex As Long

    If sourceRowIndex < 1 Or sourceRowIndex > sourceTbl.DataBodyRange.Rows.Count Then Exit Sub
    If Not QuarterPlanRowHasTask(sourceTbl, sourceRowIndex) Then Exit Sub

    targetRowIndex = FirstEmptyQuarterPlanRow(targetTbl)
    If targetRowIndex = 0 Then
        MsgBox "В целевой секции нет свободных строк.", vbExclamation
        Exit Sub
    End If

    targetTbl.DataBodyRange.Cells(targetRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value = _
        sourceTbl.DataBodyRange.Cells(sourceRowIndex, PLAN_COL_DATA_FIRST).Resize(1, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).Value

    ClearQuarterPlanRowSchedule targetTbl.DataBodyRange.Rows(targetRowIndex)
    CompactQuarterPlanTable sourceTbl, sourceRowIndex
    RefreshQuarterPlanTable sourceTbl
    RefreshQuarterPlanTable targetTbl
    SyncQuarterPlanTableActionButtons sourceTbl
    SyncQuarterPlanTableActionButtons targetTbl
End Sub

Private Sub MoveAllQuarterPlanTasks(ByVal ws As Worksheet, ByVal sourceTbl As ListObject, ByVal targetTbl As ListObject)
    Dim sourceRowIndex As Long

    Do
        sourceRowIndex = FirstTaskQuarterPlanRow(sourceTbl)
        If sourceRowIndex = 0 Then Exit Do
        If FirstEmptyQuarterPlanRow(targetTbl) = 0 Then Exit Do
        MoveQuarterPlanTask ws, sourceTbl, targetTbl, sourceRowIndex
    Loop
End Sub

Private Function MoveAllQuarterPlanTasksFast(ByVal sourceTbl As ListObject, ByVal targetTbl As ListObject) As Long
    Dim sourceRowIndex As Long
    Dim targetRowIndex As Long
    Dim dataWidth As Long
    Dim sourceValues As Variant
    Dim targetValues() As Variant
    Dim remainingValues() As Variant
    Dim taskCount As Long
    Dim remainingCount As Long
    Dim availableTargetRows As Long
    Dim moveIndex As Long
    Dim remainingIndex As Long
    Dim col As Long

    If sourceTbl.DataBodyRange Is Nothing Then Exit Function
    If targetTbl.DataBodyRange Is Nothing Then Exit Function

    targetRowIndex = FirstEmptyQuarterPlanRow(targetTbl)
    If targetRowIndex = 0 Then Exit Function

    dataWidth = PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1
    availableTargetRows = targetTbl.DataBodyRange.Rows.Count - targetRowIndex + 1
    sourceValues = sourceTbl.DataBodyRange.Cells(1, PLAN_COL_DATA_FIRST).Resize(sourceTbl.DataBodyRange.Rows.Count, dataWidth).Value

    For sourceRowIndex = 1 To sourceTbl.DataBodyRange.Rows.Count
        If QuarterPlanArrayRowHasTask(sourceValues, sourceRowIndex, PLAN_COL_QA - PLAN_COL_DATA_FIRST + 1) Then
            taskCount = taskCount + 1
        End If
    Next sourceRowIndex

    If taskCount = 0 Then Exit Function

    MoveAllQuarterPlanTasksFast = taskCount
    If MoveAllQuarterPlanTasksFast > availableTargetRows Then MoveAllQuarterPlanTasksFast = availableTargetRows
    remainingCount = taskCount - MoveAllQuarterPlanTasksFast

    If MoveAllQuarterPlanTasksFast > 0 Then ReDim targetValues(1 To MoveAllQuarterPlanTasksFast, 1 To dataWidth)
    If remainingCount > 0 Then ReDim remainingValues(1 To remainingCount, 1 To dataWidth)

    For sourceRowIndex = 1 To sourceTbl.DataBodyRange.Rows.Count
        If QuarterPlanArrayRowHasTask(sourceValues, sourceRowIndex, PLAN_COL_QA - PLAN_COL_DATA_FIRST + 1) Then
            If moveIndex < MoveAllQuarterPlanTasksFast Then
                moveIndex = moveIndex + 1
                For col = 1 To dataWidth
                    targetValues(moveIndex, col) = sourceValues(sourceRowIndex, col)
                Next col
            Else
                remainingIndex = remainingIndex + 1
                For col = 1 To dataWidth
                    remainingValues(remainingIndex, col) = sourceValues(sourceRowIndex, col)
                Next col
            End If
        End If
    Next sourceRowIndex

    If MoveAllQuarterPlanTasksFast > 0 Then
        targetTbl.DataBodyRange.Cells(targetRowIndex, PLAN_COL_DATA_FIRST).Resize(MoveAllQuarterPlanTasksFast, dataWidth).Value = targetValues
        targetTbl.DataBodyRange.Cells(targetRowIndex, PLAN_COL_READY_DATE).Resize(MoveAllQuarterPlanTasksFast, 1).ClearContents
        targetTbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & (targetTbl.DataBodyRange.Row + targetRowIndex - 1) & ":" & PLAN_TECH_LAST_COL_LETTER & (targetTbl.DataBodyRange.Row + targetRowIndex + MoveAllQuarterPlanTasksFast - 2)).ClearContents
    End If

    RewriteQuarterPlanSourceRows sourceTbl, remainingValues, remainingCount
End Function

Private Function QuarterPlanArrayRowHasTask(ByRef values As Variant, ByVal rowIndex As Long, ByVal lastTaskCol As Long) As Boolean
    Dim col As Long

    For col = 1 To lastTaskCol
        If Trim$(CStr(values(rowIndex, col))) <> "" Then
            QuarterPlanArrayRowHasTask = True
            Exit Function
        End If
    Next col
End Function

Private Sub RewriteQuarterPlanSourceRows(ByVal tbl As ListObject, ByRef remainingValues() As Variant, ByVal remainingCount As Long)
    Dim dataWidth As Long

    dataWidth = PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1

    If remainingCount > 0 Then
        tbl.DataBodyRange.Cells(1, PLAN_COL_DATA_FIRST).Resize(remainingCount, dataWidth).Value = remainingValues
        tbl.DataBodyRange.Cells(1, PLAN_COL_READY_DATE).Resize(remainingCount, 1).ClearContents
        tbl.Parent.Range(PLAN_TECH_FIRST_COL_LETTER & tbl.DataBodyRange.Row & ":" & PLAN_TECH_LAST_COL_LETTER & (tbl.DataBodyRange.Row + remainingCount - 1)).ClearContents
    End If

    If remainingCount < tbl.DataBodyRange.Rows.Count Then
        tbl.DataBodyRange.Cells(remainingCount + 1, PLAN_COL_DATA_FIRST).Resize(tbl.DataBodyRange.Rows.Count - remainingCount, dataWidth).ClearContents
    End If
End Sub

Private Function FirstTaskQuarterPlanRow(ByVal tbl As ListObject) As Long
    Dim rowIndex As Long

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        If QuarterPlanRowHasTask(tbl, rowIndex) Then
            FirstTaskQuarterPlanRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function FirstEmptyQuarterPlanRow(ByVal tbl As ListObject) As Long
    Dim rowIndex As Long

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        If Not QuarterPlanRowHasTask(tbl, rowIndex) Then
            FirstEmptyQuarterPlanRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function PreviousTaskQuarterPlanRow(ByVal tbl As ListObject, ByVal sourceRowIndex As Long) As Long
    Dim rowIndex As Long

    For rowIndex = sourceRowIndex - 1 To 1 Step -1
        If QuarterPlanRowHasTask(tbl, rowIndex) Then
            PreviousTaskQuarterPlanRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function NextTaskQuarterPlanRow(ByVal tbl As ListObject, ByVal sourceRowIndex As Long) As Long
    Dim rowIndex As Long

    For rowIndex = sourceRowIndex + 1 To tbl.DataBodyRange.Rows.Count
        If QuarterPlanRowHasTask(tbl, rowIndex) Then
            NextTaskQuarterPlanRow = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function QuarterPlanRowHasTask(ByVal tbl As ListObject, ByVal rowIndex As Long) As Boolean
    Dim col As Long
    Dim valueText As String

    For col = PLAN_COL_DATA_FIRST To PLAN_COL_QA
        valueText = Trim$(CStr(tbl.DataBodyRange.Cells(rowIndex, col).Value))
        If valueText <> "" Then
            QuarterPlanRowHasTask = True
            Exit Function
        End If
    Next col
End Function

Private Sub CompactQuarterPlanTable(ByVal tbl As ListObject, ByVal removedRowIndex As Long)
    Dim rowCount As Long
    Dim dataWidth As Long

    rowCount = tbl.DataBodyRange.Rows.Count
    dataWidth = PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1

    If removedRowIndex < rowCount Then
        tbl.DataBodyRange.Cells(removedRowIndex, PLAN_COL_DATA_FIRST).Resize(rowCount - removedRowIndex, dataWidth).Value = _
            tbl.DataBodyRange.Cells(removedRowIndex + 1, PLAN_COL_DATA_FIRST).Resize(rowCount - removedRowIndex, dataWidth).Value
    End If

    tbl.DataBodyRange.Cells(rowCount, PLAN_COL_DATA_FIRST).Resize(1, dataWidth).ClearContents
End Sub

Public Sub ReloadQuarterPlanBacklogFromEstimates(Optional ByVal confirmReset As Boolean = True)
    On Error GoTo Fail

    Dim wsEstimate As Worksheet
    Dim wsPlan As Worksheet
    Dim estimateTbl As ListObject
    Dim activeTbl As ListObject
    Dim greyTbl As ListObject
    Dim backlogTbl As ListObject
    Dim skippedCount As Long
    Dim oldEnableEvents As Boolean
    Dim oldScreenUpdating As Boolean

    If confirmReset Then
        If MsgBox(QuarterPlanTextReloadConfirm(), vbQuestion + vbYesNo) <> vbYes Then Exit Sub
    End If

    oldEnableEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    Set wsEstimate = GetTaskEstimatesWorksheet()
    Set wsPlan = GetQuarterPlanWorksheet()
    If wsEstimate Is Nothing Then Err.Raise 9, , "Task estimates worksheet not found by tblTaskEstimates."
    If wsPlan Is Nothing Then Err.Raise 9, , "Quarter plan worksheet not found by tblPlanActive."
    Set estimateTbl = wsEstimate.ListObjects("tblTaskEstimates")
    Set activeTbl = wsPlan.ListObjects("tblPlanActive")
    Set greyTbl = wsPlan.ListObjects("tblPlanGrey")
    Set backlogTbl = wsPlan.ListObjects("tblPlanBacklog")

    ClearQuarterPlanTaskData activeTbl
    ClearQuarterPlanTaskData greyTbl
    ClearQuarterPlanTaskData backlogTbl
    skippedCount = LoadEstimateRowsToBacklogFast(estimateTbl, backlogTbl)

    RefreshQuarterPlanTable backlogTbl

    If skippedCount > 0 Then
        MsgBox QuarterPlanTextBacklogFull() & CStr(skippedCount), vbExclamation
    End If

SafeExit:
    Application.ScreenUpdating = oldScreenUpdating
    Application.EnableEvents = oldEnableEvents
    Exit Sub

Fail:
    MsgBox "Ошибка ReloadQuarterPlanBacklogFromEstimates: " & CStr(Err.Number) & " - " & Err.Description, vbExclamation
    Resume SafeExit
End Sub

Private Sub ClearQuarterPlanTaskData(ByVal tbl As ListObject)
    Dim rowCount As Long

    If tbl.DataBodyRange Is Nothing Then Exit Sub

    rowCount = tbl.DataBodyRange.Rows.Count
    tbl.DataBodyRange.Cells(1, 1).Resize(rowCount, PLAN_COL_ACTION_DOWN).ClearContents
    tbl.DataBodyRange.Cells(1, PLAN_COL_DATA_FIRST).Resize(rowCount, PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1).ClearContents

    ClearQuarterPlanSchedule tbl
End Sub

Private Function LoadEstimateRowsToBacklogFast(ByVal sourceTbl As ListObject, ByVal backlogTbl As ListObject) As Long
    Dim sourceValues As Variant
    Dim levelValues As Variant
    Dim taskIdValues As Variant
    Dim parentIdValues As Variant
    Dim targetValues() As Variant
    Dim hasChildren() As Boolean
    Dim sourceRowCount As Long
    Dim targetRowCount As Long
    Dim dataWidth As Long
    Dim sourceRowIndex As Long
    Dim targetRowIndex As Long
    Dim parentRowIndex As Long
    Dim levelValue As Long
    Dim anValue As Double
    Dim beValue As Double
    Dim feValue As Double
    Dim qaValue As Double

    If sourceTbl.DataBodyRange Is Nothing Then Exit Function
    If backlogTbl.DataBodyRange Is Nothing Then Exit Function

    sourceRowCount = sourceTbl.DataBodyRange.Rows.Count
    targetRowCount = backlogTbl.DataBodyRange.Rows.Count
    dataWidth = PLAN_COL_DATA_LAST - PLAN_COL_DATA_FIRST + 1

    sourceValues = sourceTbl.DataBodyRange.Value
    levelValues = sourceTbl.Parent.Range(sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row, EST_SHEET_COL_LEVEL), _
        sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row + sourceRowCount - 1, EST_SHEET_COL_LEVEL)).Value
    taskIdValues = sourceTbl.Parent.Range(sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row, EST_SHEET_COL_TASK_ID), _
        sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row + sourceRowCount - 1, EST_SHEET_COL_TASK_ID)).Value
    parentIdValues = sourceTbl.Parent.Range(sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row, EST_SHEET_COL_PARENT_TASK_ID), _
        sourceTbl.Parent.Cells(sourceTbl.DataBodyRange.Row + sourceRowCount - 1, EST_SHEET_COL_PARENT_TASK_ID)).Value
    ReDim hasChildren(1 To sourceRowCount)
    ReDim targetValues(1 To targetRowCount, 1 To dataWidth)

    For sourceRowIndex = 1 To sourceRowCount
        If TaskEstimateArrayRowHasTask(sourceValues, sourceRowIndex) Then
            parentRowIndex = TaskEstimateArrayRowById(taskIdValues, CStr(parentIdValues(sourceRowIndex, 1)), sourceRowCount)
            If parentRowIndex > 0 Then hasChildren(parentRowIndex) = True
        End If
    Next sourceRowIndex

    For sourceRowIndex = 1 To sourceRowCount
        levelValue = TaskEstimateArrayLevel(levelValues(sourceRowIndex, 1))
        If levelValue = 0 And TaskEstimateArrayRowHasTask(sourceValues, sourceRowIndex) Then
            If targetRowIndex < targetRowCount Then
                targetRowIndex = targetRowIndex + 1
                targetValues(targetRowIndex, PLAN_COL_PRIORITY - PLAN_COL_DATA_FIRST + 1) = sourceValues(sourceRowIndex, EST_COL_PRIORITY)
                targetValues(targetRowIndex, PLAN_COL_PRIORITY - PLAN_COL_DATA_FIRST + 2) = sourceValues(sourceRowIndex, EST_COL_DIRECTION)
                targetValues(targetRowIndex, PLAN_COL_PRIORITY - PLAN_COL_DATA_FIRST + 3) = sourceValues(sourceRowIndex, EST_COL_DESCRIPTION)
                targetValues(targetRowIndex, PLAN_COL_PRIORITY - PLAN_COL_DATA_FIRST + 4) = sourceValues(sourceRowIndex, EST_COL_TICKET)
                targetValues(targetRowIndex, PLAN_COL_PRIORITY - PLAN_COL_DATA_FIRST + 5) = sourceValues(sourceRowIndex, EST_COL_NOTE)
                TaskEstimateArrayEffectiveEstimates sourceValues, levelValues, hasChildren, sourceRowIndex, sourceRowCount, anValue, beValue, feValue, qaValue
                If anValue <> 0 Then targetValues(targetRowIndex, PLAN_COL_AN - PLAN_COL_DATA_FIRST + 1) = anValue
                If beValue <> 0 Then targetValues(targetRowIndex, PLAN_COL_BE - PLAN_COL_DATA_FIRST + 1) = beValue
                If feValue <> 0 Then targetValues(targetRowIndex, PLAN_COL_FE - PLAN_COL_DATA_FIRST + 1) = feValue
                If qaValue <> 0 Then targetValues(targetRowIndex, PLAN_COL_QA - PLAN_COL_DATA_FIRST + 1) = qaValue
                targetValues(targetRowIndex, PLAN_COL_COMMENT - PLAN_COL_DATA_FIRST + 1) = sourceValues(sourceRowIndex, EST_COL_COMMENT)
            Else
                LoadEstimateRowsToBacklogFast = LoadEstimateRowsToBacklogFast + 1
            End If
        End If
    Next sourceRowIndex

    backlogTbl.DataBodyRange.Cells(1, PLAN_COL_DATA_FIRST).Resize(targetRowCount, dataWidth).Value = targetValues
    ClearQuarterPlanSchedule backlogTbl
End Function

Private Function TaskEstimateArrayRowById(ByRef taskIdValues As Variant, ByVal taskId As String, ByVal rowCount As Long) As Long
    Dim rowIndex As Long
    Dim normalizedTaskId As String

    normalizedTaskId = Trim$(taskId)
    If normalizedTaskId = "" Then Exit Function

    For rowIndex = 1 To rowCount
        If Trim$(CStr(taskIdValues(rowIndex, 1))) = normalizedTaskId Then
            TaskEstimateArrayRowById = rowIndex
            Exit Function
        End If
    Next rowIndex
End Function

Private Function TaskEstimateArrayLevel(ByVal rawValue As Variant) As Long
    Dim result As Long

    If IsNumeric(rawValue) Then result = CLng(rawValue)
    If result < 0 Then result = 0
    If result > EST_MAX_DECOMPOSITION_LEVEL Then result = EST_MAX_DECOMPOSITION_LEVEL
    TaskEstimateArrayLevel = result
End Function

Private Function TaskEstimateArrayRowHasTask(ByRef values As Variant, ByVal rowIndex As Long) As Boolean
    TaskEstimateArrayRowHasTask = TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_PRIORITY) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_DIRECTION) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_DESCRIPTION) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_TICKET) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_NOTE) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_AN) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_BE) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_FE) _
        Or TaskEstimateArrayCellHasValue(values, rowIndex, EST_COL_QA)
End Function

Private Function TaskEstimateArrayCellHasValue(ByRef values As Variant, ByVal rowIndex As Long, ByVal colIndex As Long) As Boolean
    TaskEstimateArrayCellHasValue = (Trim$(CStr(values(rowIndex, colIndex))) <> "")
End Function

Private Sub TaskEstimateArrayEffectiveEstimates(ByRef values As Variant, ByRef levelValues As Variant, ByRef hasChildren() As Boolean, ByVal rootRowIndex As Long, ByVal rowCount As Long, ByRef anValue As Double, ByRef beValue As Double, ByRef feValue As Double, ByRef qaValue As Double)
    Dim scanRow As Long
    Dim scanLevel As Long

    anValue = 0
    beValue = 0
    feValue = 0
    qaValue = 0

    If Not hasChildren(rootRowIndex) Then
        anValue = NzDouble(values(rootRowIndex, EST_COL_AN))
        beValue = NzDouble(values(rootRowIndex, EST_COL_BE))
        feValue = NzDouble(values(rootRowIndex, EST_COL_FE))
        qaValue = NzDouble(values(rootRowIndex, EST_COL_QA))
        Exit Sub
    End If

    For scanRow = rootRowIndex + 1 To rowCount
        If Not TaskEstimateArrayRowHasTask(values, scanRow) Then Exit For
        scanLevel = TaskEstimateArrayLevel(levelValues(scanRow, 1))
        If scanLevel = 0 Then Exit For
        If Not hasChildren(scanRow) Then
            anValue = anValue + NzDouble(values(scanRow, EST_COL_AN))
            beValue = beValue + NzDouble(values(scanRow, EST_COL_BE))
            feValue = feValue + NzDouble(values(scanRow, EST_COL_FE))
            qaValue = qaValue + NzDouble(values(scanRow, EST_COL_QA))
        End If
    Next scanRow
End Sub

Private Sub RefreshQuarterPlanTable(ByVal tbl As ListObject)
    Dim rowIndex As Long
    Dim rowCount As Long
    Dim actionToPlan As String
    Dim actionToGrey As String
    Dim actionToBacklog As String
    Dim actionUp As String
    Dim actionDown As String
    Dim actionValues() As Variant
    Dim dataValues As Variant
    Dim dataCol As Long
    Dim rowHasTask As Boolean
    Dim previousTaskRow As Long
    Dim actionRange As Range
    Dim effortRange As Range

    If tbl.DataBodyRange Is Nothing Then Exit Sub

    rowCount = tbl.DataBodyRange.Rows.Count
    ReDim actionValues(1 To rowCount, 1 To PLAN_COL_ACTION_DOWN)
    dataValues = tbl.DataBodyRange.Cells(1, PLAN_COL_DATA_FIRST).Resize(rowCount, PLAN_COL_QA - PLAN_COL_DATA_FIRST + 1).Value

    actionToPlan = QuarterPlanTextAddToPlan()
    actionToGrey = QuarterPlanTextToGrey()
    actionToBacklog = QuarterPlanTextToBacklog()
    actionUp = QuarterPlanTextMoveUp()
    actionDown = QuarterPlanTextMoveDown()

    For rowIndex = 1 To rowCount
        rowHasTask = False
        For dataCol = 1 To UBound(dataValues, 2)
            If Trim$(CStr(dataValues(rowIndex, dataCol))) <> "" Then
                rowHasTask = True
                Exit For
            End If
        Next dataCol

        If rowHasTask Then
            Select Case tbl.Name
                Case "tblPlanActive"
                    actionValues(rowIndex, PLAN_COL_ACTION_TO_GREY) = actionToGrey
                    actionValues(rowIndex, PLAN_COL_ACTION_TO_BACKLOG) = actionToBacklog
                Case "tblPlanGrey"
                    actionValues(rowIndex, PLAN_COL_ACTION_TO_BACKLOG) = actionToBacklog
                Case "tblPlanBacklog"
                    actionValues(rowIndex, PLAN_COL_ACTION_TO_PLAN) = actionToPlan
            End Select

            If previousTaskRow > 0 Then
                actionValues(rowIndex, PLAN_COL_ACTION_UP) = actionUp
                actionValues(previousTaskRow, PLAN_COL_ACTION_DOWN) = actionDown
            End If
            previousTaskRow = rowIndex
        End If
    Next rowIndex

    Set actionRange = tbl.DataBodyRange.Cells(1, 1).Resize(rowCount, PLAN_COL_ACTION_DOWN)
    actionRange.Value = actionValues
    actionRange.Font.Name = DESIGN_FONT_NAME
    actionRange.Font.Size = DESIGN_FONT_SIZE
    actionRange.Font.Color = RGB(255, 255, 255)
    actionRange.Font.Bold = True
    actionRange.Font.Underline = False
    actionRange.HorizontalAlignment = xlCenter
    actionRange.VerticalAlignment = xlCenter

    Set effortRange = tbl.ListColumns(PLAN_COL_EFFORT).DataBodyRange

End Sub

Public Sub RecalculateQuarterPlan()
    On Error GoTo SafeExit

    Dim wsPlan As Worksheet
    Dim wsQuarter As Worksheet
    Dim wsRefs As Worksheet
    Dim wsCapacity As Worksheet
    Dim wsSettings As Worksheet
    Dim activeTbl As ListObject
    Dim greyTbl As ListObject
    Dim backlogTbl As ListObject
    Dim holidays As Collection
    Dim firstWorkDate As Date
    Dim focusFactor As Double
    Dim rowIndex As Long
    Dim diagnostic As String

    Application.EnableEvents = False

    Set wsPlan = GetQuarterPlanWorksheet()
    Set wsQuarter = GetQuarterSettingsWorksheet()
    Set wsRefs = GetRefsWorksheet()
    Set wsSettings = FindWorksheetByTableName("tblTeamMembers")
    Set wsCapacity = ThisWorkbook.Worksheets(SHEET_CAPACITY)
    If wsPlan Is Nothing Then Err.Raise 9, , "Quarter plan worksheet not found by tblPlanActive."
    If wsQuarter Is Nothing Then Err.Raise 9, , "Quarter settings worksheet not found by tblHolidays."
    If wsRefs Is Nothing Then Err.Raise 9, , "References worksheet not found by tblPlanningRuleSettings."
    If wsSettings Is Nothing Then Err.Raise 9, , "Settings worksheet not found by tblTeamMembers."
    ApplyQuarterPlanDateFormats wsPlan
    Set activeTbl = wsPlan.ListObjects("tblPlanActive")
    Set greyTbl = wsPlan.ListObjects("tblPlanGrey")
    Set backlogTbl = wsPlan.ListObjects("tblPlanBacklog")

    ClearQuarterPlanSchedule activeTbl
    ClearQuarterPlanSchedule greyTbl
    ClearQuarterPlanSchedule backlogTbl

    If Not IsDate(wsQuarter.Range("B4").Value) Then
        MarkQuarterPlanDiagnostics activeTbl, "Quarter start date is missing."
        GoTo SafeExit
    End If

    Set holidays = GetQuarterHolidaySerials(wsQuarter)
    firstWorkDate = NextPlanWorkingDayOnOrAfter(CDate(wsQuarter.Range("B4").Value), holidays)
    focusFactor = NzDouble(wsCapacity.Range("E20").Value)

    If focusFactor <= 0 Then
        MarkQuarterPlanDiagnostics activeTbl, "Core team focus factor is missing or <= 0."
        GoTo SafeExit
    End If

    If Not LoadQuarterPlanRuleSettings(wsRefs, diagnostic) Then
        MarkQuarterPlanDiagnostics activeTbl, diagnostic
        GoTo SafeExit
    End If

    If Not LoadQuarterPlanResourcePools(wsSettings, firstWorkDate, diagnostic) Then
        MarkQuarterPlanDiagnostics activeTbl, diagnostic
        GoTo SafeExit
    End If

    Set mPlanCalcHolidays = holidays
    mPlanCalcFirstWorkDate = firstWorkDate
    mPlanCalcFocusFactor = focusFactor

    For rowIndex = 1 To activeTbl.DataBodyRange.Rows.Count
        If QuarterPlanRowHasTask(activeTbl, rowIndex) Then
            ScheduleQuarterPlanTaskUsingState activeTbl, rowIndex
        End If
    Next rowIndex

SafeExit:
    Set mPlanCalcHolidays = Nothing
    ResetQuarterPlanResourcePools
    Application.EnableEvents = True
End Sub

Private Function LoadQuarterPlanRuleSettings(ByVal wsRefs As Worksheet, ByRef diagnostic As String) As Boolean
    Dim tbl As ListObject
    Dim value As Double

    On Error Resume Next
    Set tbl = wsRefs.ListObjects("tblPlanningRuleSettings")
    On Error GoTo 0

    If tbl Is Nothing Then
        diagnostic = "Planning rule settings table is missing."
        Exit Function
    End If

    If Not TryReadPlanningRuleValue(tbl, "AN_BE_MIN_WORKDAYS", 0, 1000, True, value, diagnostic) Then Exit Function
    mPlanRuleAnBeMinWorkdays = CLng(value)

    If Not TryReadPlanningRuleValue(tbl, "AN_BE_COMPLETION_PERCENT", 0, 1, False, value, diagnostic) Then Exit Function
    mPlanRuleAnBeCompletionPercent = value

    If Not TryReadPlanningRuleValue(tbl, "BE_FE_MIN_WORKDAYS", 0, 1000, True, value, diagnostic) Then Exit Function
    mPlanRuleBeFeMinWorkdays = CLng(value)

    If Not TryReadPlanningRuleValue(tbl, "BE_FE_COMPLETION_PERCENT", 0, 1, False, value, diagnostic) Then Exit Function
    mPlanRuleBeFeCompletionPercent = value

    If Not TryReadPlanningRuleValue(tbl, "QA_FIRST_PART_PERCENT", 0, 1, False, value, diagnostic) Then Exit Function
    mPlanRuleQaFirstPartPercent = value

    LoadQuarterPlanRuleSettings = True
End Function

Private Function TryReadPlanningRuleValue(ByVal tbl As ListObject, ByVal ruleCode As String, ByVal minValue As Double, ByVal maxValue As Double, ByVal requireWhole As Boolean, ByRef result As Double, ByRef diagnostic As String) As Boolean
    Dim rowIndex As Long
    Dim codeText As String
    Dim rawValue As Variant
    Dim numericValue As Double

    If tbl.DataBodyRange Is Nothing Then
        diagnostic = "Planning rule settings table is empty."
        Exit Function
    End If

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        codeText = Trim$(CStr(tbl.DataBodyRange.Cells(rowIndex, 1).Value))
        If codeText = ruleCode Then
            rawValue = tbl.DataBodyRange.Cells(rowIndex, 3).Value
            If Not IsNumeric(rawValue) Then
                diagnostic = "Planning rule has non-numeric value: " & ruleCode & "."
                Exit Function
            End If

            numericValue = CDbl(rawValue)
            If numericValue < minValue Or numericValue > maxValue Then
                diagnostic = "Planning rule is outside allowed range: " & ruleCode & "."
                Exit Function
            End If

            If requireWhole And numericValue <> Fix(numericValue) Then
                diagnostic = "Planning rule must be a whole number: " & ruleCode & "."
                Exit Function
            End If

            result = numericValue
            TryReadPlanningRuleValue = True
            Exit Function
        End If
    Next rowIndex

    diagnostic = "Planning rule is missing: " & ruleCode & "."
End Function

Private Sub ScheduleQuarterPlanTaskUsingState(ByVal tbl As ListObject, ByVal rowIndex As Long)
    Dim rowRange As Range
    Dim sheetRow As Long
    Dim statusText As String
    Dim includeBe As Boolean
    Dim includeFe As Boolean
    Dim includeQa As Boolean
    Dim anEstimate As Double
    Dim beEstimate As Double
    Dim feEstimate As Double
    Dim qaEstimate As Double
    Dim anBaseDuration As Long
    Dim beBaseDuration As Long
    Dim feBaseDuration As Long
    Dim qaBaseDuration As Long
    Dim anActualDuration As Long
    Dim beActualDuration As Long
    Dim feActualDuration As Long
    Dim qaActualDuration As Long
    Dim qaFirstBaseDuration As Long
    Dim qaSecondBaseDuration As Long
    Dim totalDuration As Long
    Dim diagnostic As String
    Dim anStart As Date
    Dim anFinish As Date
    Dim anProgressReady As Date
    Dim beStart As Date
    Dim beFinish As Date
    Dim beProgressReady As Date
    Dim feStart As Date
    Dim feFinish As Date
    Dim qaStart As Date
    Dim qaFinish As Date
    Dim qaFirstStart As Date
    Dim qaFirstFinish As Date
    Dim qaFirstActualDuration As Long
    Dim qaSecondStart As Date
    Dim qaSecondFinish As Date
    Dim qaSecondActualDuration As Long
    Dim qaFirstEstimate As Double
    Dim qaSecondEstimate As Double
    Dim beReady As Date
    Dim feReady As Date
    Dim qaFirstReady As Date
    Dim qaSecondReady As Date
    Dim latestDevFinish As Date
    Dim hasDevFinish As Boolean
    Dim latestPreQaFinish As Date
    Dim hasPreQaFinish As Boolean
    Dim finalFinish As Date
    Dim hasFinalFinish As Boolean
    Dim unusedProgressReady As Date

    Set rowRange = tbl.DataBodyRange.Rows(rowIndex)
    sheetRow = rowRange.Row
    statusText = Trim$(CStr(rowRange.Cells(1, PLAN_COL_STATUS).Value))
    includeBe = (statusText <> PLAN_STATUS_ANALYSIS_READY And statusText <> PLAN_STATUS_FRONTEND_READY And statusText <> PLAN_STATUS_DEFERRED)
    includeFe = (statusText <> PLAN_STATUS_ANALYSIS_READY And statusText <> PLAN_STATUS_BACKEND_READY And statusText <> PLAN_STATUS_DEFERRED)
    includeQa = (statusText <> PLAN_STATUS_ANALYSIS_READY And statusText <> PLAN_STATUS_DEV_READY And statusText <> PLAN_STATUS_BACKEND_READY And statusText <> PLAN_STATUS_FRONTEND_READY And statusText <> PLAN_STATUS_DEFERRED)

    anEstimate = NzDouble(rowRange.Cells(1, PLAN_COL_AN).Value)
    beEstimate = NzDouble(rowRange.Cells(1, PLAN_COL_BE).Value)
    feEstimate = NzDouble(rowRange.Cells(1, PLAN_COL_FE).Value)
    qaEstimate = NzDouble(rowRange.Cells(1, PLAN_COL_QA).Value)

    anBaseDuration = PlannedDuration(anEstimate, mPlanCalcFocusFactor)
    beBaseDuration = PlannedDuration(beEstimate, mPlanCalcFocusFactor)
    feBaseDuration = PlannedDuration(feEstimate, mPlanCalcFocusFactor)
    qaBaseDuration = PlannedDuration(qaEstimate, mPlanCalcFocusFactor)
    totalDuration = anBaseDuration + beBaseDuration + feBaseDuration + qaBaseDuration
    If statusText = PLAN_STATUS_DEFERRED Then
        rowRange.Cells(1, PLAN_COL_EFFORT).Value = totalDuration
        GoTo SetReadyDate
    End If
    If Not includeBe Then beBaseDuration = 0
    If Not includeFe Then feBaseDuration = 0
    If Not includeQa Then qaBaseDuration = 0
    qaFirstBaseDuration = CeilingPositive(CDbl(qaBaseDuration) * mPlanRuleQaFirstPartPercent)
    qaSecondBaseDuration = qaBaseDuration - qaFirstBaseDuration

    rowRange.Cells(1, PLAN_COL_EFFORT).Value = totalDuration

    If anBaseDuration > 0 And mPlanCalcAnCount = 0 Then diagnostic = AppendDiagnostic(diagnostic, "No AN resources.")
    If includeBe And beBaseDuration > 0 And mPlanCalcBeCount = 0 Then diagnostic = AppendDiagnostic(diagnostic, "No BE resources.")
    If includeFe And feBaseDuration > 0 And mPlanCalcFeCount = 0 Then diagnostic = AppendDiagnostic(diagnostic, "No FE resources.")
    If includeQa And qaBaseDuration > 0 And mPlanCalcQaCount = 0 Then diagnostic = AppendDiagnostic(diagnostic, "No QA resources.")

    If diagnostic <> "" Then
        rowRange.Cells(1, PLAN_COL_READY_DATE).ClearContents
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_DIAGNOSTIC).Value = diagnostic
        Exit Sub
    End If

    If anBaseDuration > 0 Then
        AllocateStage mPlanCalcAnResources, mPlanCalcAnCount, mPlanCalcFirstWorkDate, anEstimate, mPlanCalcFocusFactor, mPlanCalcHolidays, _
            PlanProgressThreshold(anBaseDuration, mPlanRuleAnBeMinWorkdays, mPlanRuleAnBeCompletionPercent, mPlanCalcFocusFactor, anEstimate), _
            anStart, anFinish, anActualDuration, anProgressReady
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS).Value = anActualDuration
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 4).Value = anStart
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 5).Value = anFinish
        RememberLatestFinish anFinish, latestPreQaFinish, hasPreQaFinish
        If statusText = PLAN_STATUS_ANALYSIS_READY Then
            finalFinish = anFinish
            hasFinalFinish = True
            GoTo SetReadyDate
        End If
        If anBaseDuration <= mPlanRuleAnBeMinWorkdays Then
            beReady = NextPlanWorkingDayAfter(anFinish, mPlanCalcHolidays)
        Else
            beReady = anProgressReady
        End If
        If beReady = 0 Then beReady = NextPlanWorkingDayAfter(anFinish, mPlanCalcHolidays)
        qaFirstReady = NextPlanWorkingDayAfter(anFinish, mPlanCalcHolidays)
    Else
        beReady = mPlanCalcFirstWorkDate
        qaFirstReady = mPlanCalcFirstWorkDate
    End If

    If includeBe And beBaseDuration > 0 Then
        AllocateStage mPlanCalcBeResources, mPlanCalcBeCount, beReady, beEstimate, mPlanCalcFocusFactor, mPlanCalcHolidays, _
            PlanProgressThreshold(beBaseDuration, mPlanRuleBeFeMinWorkdays, mPlanRuleBeFeCompletionPercent, mPlanCalcFocusFactor, beEstimate), _
            beStart, beFinish, beActualDuration, beProgressReady
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 1).Value = beActualDuration
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 6).Value = beStart
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 7).Value = beFinish
        RememberLatestFinish beFinish, latestPreQaFinish, hasPreQaFinish
        RememberLatestFinish beFinish, latestDevFinish, hasDevFinish
    End If

    If includeFe And feBaseDuration > 0 Then
        If beBaseDuration = 0 Then
            feReady = beReady
        ElseIf beEstimate <= mPlanRuleBeFeMinWorkdays Or beBaseDuration <= mPlanRuleBeFeMinWorkdays Then
            feReady = NextPlanWorkingDayAfter(beFinish, mPlanCalcHolidays)
        Else
            feReady = beProgressReady
        End If
        If feReady = 0 Then feReady = NextPlanWorkingDayAfter(beFinish, mPlanCalcHolidays)

        unusedProgressReady = 0
        AllocateStage mPlanCalcFeResources, mPlanCalcFeCount, feReady, feEstimate, mPlanCalcFocusFactor, mPlanCalcHolidays, 0, _
            feStart, feFinish, feActualDuration, unusedProgressReady
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 2).Value = feActualDuration
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 8).Value = feStart
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 9).Value = feFinish
        RememberLatestFinish feFinish, latestPreQaFinish, hasPreQaFinish
        RememberLatestFinish feFinish, latestDevFinish, hasDevFinish
    End If

    If statusText = PLAN_STATUS_DEV_READY Or statusText = PLAN_STATUS_BACKEND_READY Or statusText = PLAN_STATUS_FRONTEND_READY Then
        If hasDevFinish Then
            finalFinish = latestDevFinish
            hasFinalFinish = True
        ElseIf hasPreQaFinish Then
            finalFinish = latestPreQaFinish
            hasFinalFinish = True
        End If
        GoTo SetReadyDate
    End If

    If includeQa And qaBaseDuration > 0 Then
        qaFirstEstimate = EstimateForBaseDurationPart(qaFirstBaseDuration, qaEstimate, mPlanCalcFocusFactor)
        qaSecondEstimate = qaEstimate - qaFirstEstimate
        If qaSecondEstimate < 0.000001 Then qaSecondEstimate = 0

        If qaFirstEstimate > 0 Then
            unusedProgressReady = 0
            AllocateStage mPlanCalcQaResources, mPlanCalcQaCount, qaFirstReady, qaFirstEstimate, mPlanCalcFocusFactor, mPlanCalcHolidays, 0, _
                qaFirstStart, qaFirstFinish, qaFirstActualDuration, unusedProgressReady
            qaStart = qaFirstStart
            qaFinish = qaFirstFinish
            qaActualDuration = qaActualDuration + qaFirstActualDuration
        End If

        If hasDevFinish Then
            qaSecondReady = NextPlanWorkingDayAfter(latestDevFinish, mPlanCalcHolidays)
        ElseIf hasPreQaFinish Then
            qaSecondReady = NextPlanWorkingDayAfter(latestPreQaFinish, mPlanCalcHolidays)
        Else
            qaSecondReady = mPlanCalcFirstWorkDate
        End If

        If qaSecondEstimate > 0 Then
            unusedProgressReady = 0
            AllocateStage mPlanCalcQaResources, mPlanCalcQaCount, qaSecondReady, qaSecondEstimate, mPlanCalcFocusFactor, mPlanCalcHolidays, 0, _
                qaSecondStart, qaSecondFinish, qaSecondActualDuration, unusedProgressReady
            If qaStart = 0 Or qaSecondStart < qaStart Then qaStart = qaSecondStart
            If qaSecondFinish > qaFinish Then qaFinish = qaSecondFinish
            qaActualDuration = qaActualDuration + qaSecondActualDuration
        End If

        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 3).Value = qaActualDuration
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 10).Value = qaStart
        tbl.Parent.Cells(sheetRow, PLAN_SHEET_COL_AN_DAYS + 11).Value = qaFinish
        finalFinish = qaFinish
        hasFinalFinish = True
    ElseIf hasPreQaFinish Then
        finalFinish = latestPreQaFinish
        hasFinalFinish = True
    End If

SetReadyDate:
    If hasFinalFinish Then
        rowRange.Cells(1, PLAN_COL_READY_DATE).Value = finalFinish
    Else
        rowRange.Cells(1, PLAN_COL_READY_DATE).ClearContents
    End If
End Sub

Private Function NzDouble(ByVal value As Variant) As Double
    If IsError(value) Or IsEmpty(value) Or Trim$(CStr(value)) = "" Then
        NzDouble = 0
    ElseIf IsNumeric(value) Then
        NzDouble = CDbl(value)
    Else
        NzDouble = 0
    End If
End Function

Private Function PlannedDuration(ByVal estimate As Double, ByVal focusFactor As Double) As Long
    If estimate <= 0 Or focusFactor <= 0 Then
        PlannedDuration = 0
    Else
        PlannedDuration = CeilingPositive(estimate / focusFactor)
    End If
End Function

Private Function CeilingPositive(ByVal value As Double) As Long
    Dim floorValue As Long

    If value <= 0 Then
        CeilingPositive = 0
    ElseIf value < 1 Then
        CeilingPositive = 1
    Else
        floorValue = CLng(Fix(value))
        If value - CDbl(floorValue) <= 0.000001 Then
            CeilingPositive = floorValue
        Else
            CeilingPositive = floorValue + 1
        End If
    End If
End Function

Private Function EstimateForBaseDurationPart(ByVal partDuration As Long, ByVal totalEstimate As Double, ByVal focusFactor As Double) As Double
    EstimateForBaseDurationPart = CDbl(partDuration) * focusFactor
    If EstimateForBaseDurationPart > totalEstimate Then EstimateForBaseDurationPart = totalEstimate
End Function

Private Function PlanProgressThreshold(ByVal baseDuration As Long, ByVal minWorkdays As Long, ByVal completionPercent As Double, ByVal focusFactor As Double, ByVal estimate As Double) As Double
    Dim minThreshold As Double
    Dim percentThreshold As Double
    Dim percentWorkdays As Long

    If minWorkdays > 0 Then minThreshold = CDbl(minWorkdays) * focusFactor
    percentWorkdays = CeilingPositive(CDbl(baseDuration) * completionPercent)
    If percentWorkdays > 0 Then percentThreshold = CDbl(percentWorkdays) * focusFactor

    If minThreshold >= percentThreshold Then
        PlanProgressThreshold = minThreshold
    Else
        PlanProgressThreshold = percentThreshold
    End If
    If PlanProgressThreshold > estimate Then PlanProgressThreshold = estimate
End Function

Private Sub ResetQuarterPlanResourcePools()
    mPlanCalcAnCount = 0
    mPlanCalcBeCount = 0
    mPlanCalcFeCount = 0
    mPlanCalcQaCount = 0
    Erase mPlanCalcAnResources
    Erase mPlanCalcBeResources
    Erase mPlanCalcFeResources
    Erase mPlanCalcQaResources
End Sub

Private Function LoadQuarterPlanResourcePools(ByVal wsSettings As Worksheet, ByVal firstWorkDate As Date, ByRef diagnostic As String) As Boolean
    Dim tbl As ListObject
    Dim rowIndex As Long
    Dim rowRange As Range
    Dim roleText As String
    Dim allocation As Double

    ResetQuarterPlanResourcePools

    On Error Resume Next
    Set tbl = wsSettings.ListObjects("tblTeamMembers")
    On Error GoTo 0

    If tbl Is Nothing Then
        diagnostic = "Team members table is missing."
        Exit Function
    End If

    If tbl.DataBodyRange Is Nothing Then
        LoadQuarterPlanResourcePools = True
        Exit Function
    End If

    For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
        Set rowRange = tbl.DataBodyRange.Rows(rowIndex)
        roleText = Trim$(CStr(rowRange.Cells(1, TEAM_MEMBER_COL_ROLE).Value))
        allocation = NzDouble(rowRange.Cells(1, TEAM_MEMBER_COL_ALLOCATION).Value)
        If roleText <> "" And allocation > 0 Then
            If IsPlanRoleForExpertise(roleText, "AN") Then
                AddQuarterPlanResource mPlanCalcAnResources, mPlanCalcAnCount, allocation, firstWorkDate, rowRange
            ElseIf IsPlanRoleForExpertise(roleText, "BE") Then
                AddQuarterPlanResource mPlanCalcBeResources, mPlanCalcBeCount, allocation, firstWorkDate, rowRange
            ElseIf IsPlanRoleForExpertise(roleText, "FE") Then
                AddQuarterPlanResource mPlanCalcFeResources, mPlanCalcFeCount, allocation, firstWorkDate, rowRange
            ElseIf IsPlanRoleForExpertise(roleText, "QA") Then
                AddQuarterPlanResource mPlanCalcQaResources, mPlanCalcQaCount, allocation, firstWorkDate, rowRange
            End If
        End If
    Next rowIndex

    LoadQuarterPlanResourcePools = True
End Function

Private Sub AddQuarterPlanResource(ByRef resources() As PlanResourceState, ByRef resourceCount As Long, ByVal allocation As Double, ByVal firstWorkDate As Date, ByVal rowRange As Range)
    resourceCount = resourceCount + 1
    If resourceCount = 1 Then
        ReDim resources(1 To 1)
    Else
        ReDim Preserve resources(1 To resourceCount)
    End If

    With resources(resourceCount)
        .Allocation = allocation
        .NextAvailable = firstWorkDate
    End With
    AssignPlanResourceVacation resources(resourceCount), 1, rowRange.Cells(1, 4).Value, rowRange.Cells(1, 5).Value
    AssignPlanResourceVacation resources(resourceCount), 2, rowRange.Cells(1, 7).Value, rowRange.Cells(1, 8).Value
    AssignPlanResourceVacation resources(resourceCount), 3, rowRange.Cells(1, 10).Value, rowRange.Cells(1, 11).Value
End Sub

Private Sub AssignPlanResourceVacation(ByRef resource As PlanResourceState, ByVal periodIndex As Long, ByVal startValue As Variant, ByVal endValue As Variant)
    Dim startDate As Date
    Dim endDate As Date

    If Not IsDate(startValue) Then Exit Sub
    If Not IsDate(endValue) Then Exit Sub

    startDate = DateValue(CDate(startValue))
    endDate = DateValue(CDate(endValue))
    If endDate < startDate Then Exit Sub

    Select Case periodIndex
        Case 1
            resource.VacationStart1 = startDate
            resource.VacationEnd1 = endDate
        Case 2
            resource.VacationStart2 = startDate
            resource.VacationEnd2 = endDate
        Case 3
            resource.VacationStart3 = startDate
            resource.VacationEnd3 = endDate
    End Select
End Sub

Private Function IsPlanRoleForExpertise(ByVal roleText As String, ByVal expertiseCode As String) As Boolean
    Select Case expertiseCode
        Case "AN"
            IsPlanRoleForExpertise = (roleText = PlanRoleAnalyst() Or roleText = PlanRoleAnalysisLead())
        Case "BE"
            IsPlanRoleForExpertise = (roleText = PlanRoleBackendDeveloper() Or roleText = PlanRoleTeamLead())
        Case "FE"
            IsPlanRoleForExpertise = (roleText = PlanRoleFrontendDeveloper())
        Case "QA"
            IsPlanRoleForExpertise = (roleText = PlanRoleTester())
    End Select
End Function

Private Sub AllocateStage(ByRef resources() As PlanResourceState, ByVal resourceCount As Long, ByVal earliestStart As Date, ByVal estimate As Double, ByVal focusFactor As Double, ByVal holidays As Collection, ByVal progressThreshold As Double, ByRef stageStart As Date, ByRef stageFinish As Date, ByRef stageDuration As Long, ByRef progressReady As Date)
    Dim workDate As Date
    Dim resourceIndex As Long
    Dim dailyProgress As Double
    Dim completed As Double
    Dim guardDays As Long

    If estimate <= 0 Then Exit Sub
    workDate = NextPlanWorkingDayOnOrAfter(earliestStart, holidays)

    Do While completed + 0.000001 < estimate
        guardDays = guardDays + 1
        If guardDays > 20000 Then Err.Raise 5, "AllocateStage", "Resource allocation exceeded 20000 working-day checks."

        resourceIndex = FindAvailablePlanResourceIndex(resources, resourceCount, workDate, focusFactor)
        If resourceIndex > 0 Then
            If stageStart = 0 Then stageStart = workDate
            dailyProgress = resources(resourceIndex).Allocation * focusFactor
            completed = completed + dailyProgress
            If completed > estimate Then completed = estimate
            stageFinish = workDate
            resources(resourceIndex).NextAvailable = NextPlanWorkingDayAfter(workDate, holidays)
            If progressThreshold > 0 And progressReady = 0 And completed + 0.000001 >= progressThreshold Then
                progressReady = NextPlanWorkingDayAfter(workDate, holidays)
            End If
        End If
        workDate = NextPlanWorkingDayAfter(workDate, holidays)
    Loop

    If progressThreshold > 0 And progressReady = 0 Then progressReady = NextPlanWorkingDayAfter(stageFinish, holidays)
    stageDuration = PlanWorkingDaysBetween(stageStart, stageFinish, holidays)
End Sub

Private Function FindAvailablePlanResourceIndex(ByRef resources() As PlanResourceState, ByVal resourceCount As Long, ByVal workDate As Date, ByVal focusFactor As Double) As Long
    Dim resourceIndex As Long
    Dim candidateProgress As Double
    Dim bestProgress As Double

    For resourceIndex = 1 To resourceCount
        If resources(resourceIndex).NextAvailable <= workDate Then
            If Not IsPlanResourceOnVacation(resources(resourceIndex), workDate) Then
                candidateProgress = resources(resourceIndex).Allocation * focusFactor
                If candidateProgress > bestProgress Then
                    bestProgress = candidateProgress
                    FindAvailablePlanResourceIndex = resourceIndex
                End If
            End If
        End If
    Next resourceIndex
End Function

Private Function IsPlanResourceOnVacation(ByRef resource As PlanResourceState, ByVal workDate As Date) As Boolean
    If IsDateInRange(workDate, resource.VacationStart1, resource.VacationEnd1) Then
        IsPlanResourceOnVacation = True
    ElseIf IsDateInRange(workDate, resource.VacationStart2, resource.VacationEnd2) Then
        IsPlanResourceOnVacation = True
    ElseIf IsDateInRange(workDate, resource.VacationStart3, resource.VacationEnd3) Then
        IsPlanResourceOnVacation = True
    End If
End Function

Private Function IsDateInRange(ByVal value As Date, ByVal startDate As Date, ByVal endDate As Date) As Boolean
    If startDate = 0 Or endDate = 0 Then Exit Function
    IsDateInRange = (DateValue(value) >= DateValue(startDate) And DateValue(value) <= DateValue(endDate))
End Function

Private Function AddPlanWorkingDays(ByVal startDate As Date, ByVal dayCount As Long, ByVal holidays As Collection) As Date
    Dim resultDate As Date
    Dim completedDays As Long

    If dayCount <= 0 Then Exit Function

    resultDate = NextPlanWorkingDayOnOrAfter(startDate, holidays)
    completedDays = 1

    Do While completedDays < dayCount
        resultDate = NextPlanWorkingDayAfter(resultDate, holidays)
        completedDays = completedDays + 1
    Loop

    AddPlanWorkingDays = resultDate
End Function

Private Function PlanWorkingDaysBetween(ByVal startDate As Date, ByVal finishDate As Date, ByVal holidays As Collection) As Long
    Dim workDate As Date

    If startDate = 0 Or finishDate = 0 Then Exit Function
    If DateValue(finishDate) < DateValue(startDate) Then Exit Function

    workDate = NextPlanWorkingDayOnOrAfter(startDate, holidays)
    Do While workDate <= DateValue(finishDate)
        PlanWorkingDaysBetween = PlanWorkingDaysBetween + 1
        workDate = NextPlanWorkingDayAfter(workDate, holidays)
    Loop
End Function

Private Function NextPlanWorkingDayOnOrAfter(ByVal value As Date, ByVal holidays As Collection) As Date
    Dim resultDate As Date
    resultDate = DateValue(value)

    Do While Not IsPlanWorkingDay(resultDate, holidays)
        resultDate = DateAdd("d", 1, resultDate)
    Loop

    NextPlanWorkingDayOnOrAfter = resultDate
End Function

Private Function NextPlanWorkingDayAfter(ByVal value As Date, ByVal holidays As Collection) As Date
    NextPlanWorkingDayAfter = NextPlanWorkingDayOnOrAfter(DateAdd("d", 1, DateValue(value)), holidays)
End Function

Private Function IsPlanWorkingDay(ByVal value As Date, ByVal holidays As Collection) As Boolean
    If Weekday(value, vbMonday) > 5 Then Exit Function
    If IsPlanHoliday(value, holidays) Then Exit Function
    IsPlanWorkingDay = True
End Function

Private Function IsPlanHoliday(ByVal value As Date, ByVal holidays As Collection) As Boolean
    Dim index As Long
    Dim serial As Long

    serial = CLng(DateValue(value))
    For index = 1 To holidays.Count
        If CLng(holidays.Item(index)) = serial Then
            IsPlanHoliday = True
            Exit Function
        End If
    Next index
End Function

Private Function GetQuarterHolidaySerials(ByVal wsQuarter As Worksheet) As Collection
    Dim holidays As Collection
    Dim tbl As ListObject
    Dim rowIndex As Long
    Dim yesText As String

    Set holidays = New Collection
    yesText = ChrW$(&H414) & ChrW$(&H430)

    On Error Resume Next
    Set tbl = wsQuarter.ListObjects("tblHolidays")
    On Error GoTo 0

    If Not tbl Is Nothing Then
        If Not tbl.DataBodyRange Is Nothing Then
            For rowIndex = 1 To tbl.DataBodyRange.Rows.Count
                If IsDate(tbl.DataBodyRange.Cells(rowIndex, 1).Value) Then
                    If Trim$(CStr(tbl.DataBodyRange.Cells(rowIndex, 4).Value)) = yesText Then
                        holidays.Add CLng(DateValue(tbl.DataBodyRange.Cells(rowIndex, 1).Value))
                    End If
                End If
            Next rowIndex
        End If
    End If

    Set GetQuarterHolidaySerials = holidays
End Function

Private Function LaterDate(ByVal firstDate As Date, ByVal secondDate As Date) As Date
    If firstDate >= secondDate Then
        LaterDate = firstDate
    Else
        LaterDate = secondDate
    End If
End Function

Private Sub RememberLatestFinish(ByVal finishDate As Date, ByRef latestFinish As Date, ByRef hasFinish As Boolean)
    If finishDate = 0 Then Exit Sub
    If Not hasFinish Or finishDate > latestFinish Then
        latestFinish = finishDate
        hasFinish = True
    End If
End Sub

Private Sub ApplyHolidayTableFormatting(ByVal tbl As ListObject)
    Dim rowHeight As Double
    Dim yesText As String
    Dim noText As String

    If tbl.DataBodyRange Is Nothing Then Exit Sub

    yesText = ChrW$(&H414) & ChrW$(&H430)
    noText = ChrW$(&H41D) & ChrW$(&H435) & ChrW$(&H442)

    rowHeight = tbl.DataBodyRange.Rows(1).RowHeight

    tbl.DataBodyRange.Rows(1).Copy
    tbl.DataBodyRange.PasteSpecial Paste:=xlPasteFormats
    Application.CutCopyMode = False

    tbl.DataBodyRange.RowHeight = rowHeight
    ApplyDateFormat tbl.ListColumns(1).DataBodyRange
    tbl.ListColumns(2).DataBodyRange.WrapText = True
    tbl.ListColumns(3).DataBodyRange.WrapText = True
    tbl.ListColumns(4).DataBodyRange.HorizontalAlignment = xlCenter

    With tbl.ListColumns(4).DataBodyRange.Validation
        .Delete
        .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:=yesText & "," & noText
        .IgnoreBlank = True
        .InCellDropdown = True
    End With
End Sub
