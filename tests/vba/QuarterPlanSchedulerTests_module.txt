Option Explicit

Private Const QPS_TEST_SCENARIO_COUNT As Long = 63

Public Function RunQuarterPlanSchedulerUnitTests() As String
    Dim scenario As Long
    Dim completedScenarios As Long
    Dim tasks As Variant, resources As Variant, holidays As Variant, rules As Variant, result As Variant
    Dim firstDate As Date, focusFactor As Double, designFocusFactor As Double

    On Error GoTo TestFailure
    For scenario = 1 To QPS_TEST_SCENARIO_COUNT
        BuildSchedulerScenario scenario, tasks, resources, holidays, rules, firstDate, focusFactor, designFocusFactor
        result = QuarterPlanSchedule(tasks, resources, holidays, rules, firstDate, focusFactor, designFocusFactor)
        AssertSchedulerScenario scenario, result
        completedScenarios = completedScenarios + 1
    Next scenario
    RunQuarterPlanSchedulerUnitTests = "PASS|" & CStr(completedScenarios)
    Exit Function
TestFailure:
    RunQuarterPlanSchedulerUnitTests = "FAIL|" & CStr(scenario) & "|" & Err.Description
End Function

Private Sub AssertSchedulerScenario(ByVal scenario As Long, ByRef result As Variant)
    Dim expectedRows As Long, actualRows As Long, rowIndex As Long, colIndex As Long
    Dim expected As Variant

    expectedRows = SchedulerExpectedRows(scenario)
    actualRows = SchedulerMatrixRows(result)
    If actualRows <> expectedRows Then
        Err.Raise vbObjectError + 2100 + scenario, "QuarterPlanSchedulerTests", SchedulerFailurePrefix(scenario) & "row count expected " & CStr(expectedRows) & ", actual " & CStr(actualRows)
    End If
    If expectedRows = 0 Then
        If Not IsEmpty(result) Then Err.Raise vbObjectError + 2100 + scenario, "QuarterPlanSchedulerTests", SchedulerFailurePrefix(scenario) & "result expected <empty>, actual non-empty scalar"
        Exit Sub
    End If

    For rowIndex = 1 To expectedRows
        expected = SchedulerExpectedRow(scenario, rowIndex)
        For colIndex = 1 To 18
            AssertSchedulerValue scenario, rowIndex, colIndex, expected(colIndex), result(rowIndex, colIndex)
        Next colIndex
    Next rowIndex
End Sub

    If scenario = 52 Then SchedulerExpectedRows = 3 Else If scenario = 63 Then SchedulerExpectedRows = 6 Else SchedulerExpectedRows = 2
End Function

Private Sub AssertSchedulerValue(ByVal scenario As Long, ByVal rowIndex As Long, ByVal colIndex As Long, ByVal expected As Variant, ByVal actual As Variant)
    Dim matches As Boolean
    If IsEmpty(expected) Then
        matches = IsEmpty(actual)
    ElseIf SchedulerOutputIsDate(colIndex) Then
        If IsDate(actual) Then matches = (CLng(CDate(actual)) = CLng(CDate(expected)))
    ElseIf IsNumeric(expected) Then
        If IsNumeric(actual) Then matches = (Abs(CDbl(actual) - CDbl(expected)) <= 0.0000001)
    Else
        If Not IsError(actual) Then matches = (CStr(actual) = CStr(expected))
    End If
    If matches Then Exit Sub
    Err.Raise vbObjectError + 2100 + scenario, "QuarterPlanSchedulerTests", SchedulerFailurePrefix(scenario) & "row " & CStr(rowIndex) & " field " & SchedulerOutputFieldName(colIndex) & " expected " & SchedulerValueText(expected, colIndex) & ", actual " & SchedulerValueText(actual, colIndex)
End Sub

Private Function SchedulerOutputIsDate(ByVal colIndex As Long) As Boolean
    SchedulerOutputIsDate = (colIndex = 2 Or (colIndex >= 7 And colIndex <= 14) Or colIndex = 17 Or colIndex = 18)
End Function

Private Function SchedulerValueText(ByVal value As Variant, ByVal colIndex As Long) As String
    If IsError(value) Then
        SchedulerValueText = "<error>"
    ElseIf IsEmpty(value) Then
        SchedulerValueText = "<empty>"
    ElseIf SchedulerOutputIsDate(colIndex) And IsDate(value) Then
        SchedulerValueText = Format$(CDate(value), "yyyy-mm-dd")
    Else
        SchedulerValueText = CStr(value)
    End If
End Function

Private Function SchedulerMatrixRows(ByRef value As Variant) As Long
